// ─── HELPER: lấy/tạo sheet theo tên ─────────────────────────
function _sheet(name, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    if (headers) sh.appendRow(headers);
  }
  return sh;
}

function _clean(v) {
  return v instanceof Date ? v.toISOString() : v;
}

function getInitData(passwordInput) {
  const login = checkLogin(passwordInput);
  if (!login || !login.success) return { login: login };
  return {
    login: login,
    employees: getEmployeeList(),
    products: getProductsCached(),
    categories: getProductCategories()
  };
}

function _empSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName('NhanVien');
  if (!sh) {
    sh = ss.insertSheet('NhanVien');
    sh.appendRow(['Tên Hiển Thị', 'Tên Đăng Nhập', 'Email', 'Quyền']);
    sh.appendRow(['Quản Lý', '9sach2026', '', 'Admin']);
  } else {
    const hdr = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
    if (hdr.length < 3 || hdr[2] !== 'Email') sh.getRange(1, 3).setValue('Email');
    if (hdr.length < 4 || hdr[3] !== 'Quyền') {
      sh.getRange(1, 4).setValue('Quyền');
      const data = sh.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (!data[i][3]) sh.getRange(i + 1, 4).setValue(i === 1 ? 'Admin' : 'NhanVien');
      }
    }
  }
  return sh;
}

function checkLogin(passwordInput) {
  try {
    const sh = _empSheet();
    const data = sh.getDataRange().getValues();
    const clean = passwordInput.toString().trim();
    for (let i = 1; i < data.length; i++) {
      if (data[i][1].toString().trim() === clean) {
        return {
          success: true,
          displayName: data[i][0].toString().trim(),
          username: data[i][1].toString().trim(),
          email: (data[i][2] || '').toString().trim(),
          role: (data[i][3] || 'NhanVien').toString().trim()
        };
      }
    }
    return { success: false };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function saveNewEmployee(displayName, username, email, role) {
  try {
    const sh = _empSheet();
    const lastRow = sh.getLastRow();
    const usernames = lastRow > 1 ? sh.getRange(2, 2, lastRow - 1, 1).getValues() : [];
    const cleanU = username.toString().trim();
    const cleanD = displayName.toString().trim();
    if (!cleanU || !cleanD) return 'Vui lòng nhập đầy đủ thông tin!';
    for (let i = 0; i < usernames.length; i++) {
      if (usernames[i][0].toString().trim() === cleanU) return 'Lỗi: Tên đăng nhập này đã tồn tại!';
    }
    sh.appendRow([cleanD, cleanU, email || '', role || 'NhanVien']);
    return 'Thêm nhân viên mới thành công!';
  } catch (e) {
    return 'Lỗi thêm nhân viên: ' + e.toString();
  }
}
function updateEmployeeRecord(oldUsername, data) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch (e) { return { success: false, error: 'Hệ thống đang bận, thử lại sau.' }; }
  try {
    const sh = _empSheet();
    const lastRow = sh.getLastRow();
    if (lastRow < 2) return { success: false, error: 'Không có nhân viên nào.' };
    const rows = sh.getRange(2, 1, lastRow - 1, 4).getValues();
    const cleanOld = (oldUsername || '').toString().trim();
    const newUsername = (data.username || '').toString().trim();
    const newDisplayName = (data.displayName || '').toString().trim();
    if (!newUsername || !newDisplayName) return { success: false, error: 'Thiếu tên hiển thị hoặc tên đăng nhập.' };

    let targetIdx = -1;
    for (let i = 0; i < rows.length; i++) {
      if (rows[i][1].toString().trim() === cleanOld) { targetIdx = i; break; }
    }
    if (targetIdx === -1) return { success: false, error: 'Không tìm thấy nhân viên này.' };

    for (let i = 0; i < rows.length; i++) {
      if (i !== targetIdx && rows[i][1].toString().trim() === newUsername) {
        return { success: false, error: 'Tên đăng nhập này đã được dùng bởi nhân viên khác.' };
      }
    }

    const oldRole = (rows[targetIdx][3] || 'NhanVien').toString();
    if (oldRole === 'Admin' && data.role !== 'Admin') {
      const adminCount = rows.filter(r => (r[3] || '') === 'Admin').length;
      if (adminCount <= 1) return { success: false, error: 'Không thể hạ quyền Admin duy nhất còn lại.' };
    }

    sh.getRange(targetIdx + 2, 1, 1, 4).setValues([[newDisplayName, newUsername, data.email || '', data.role || 'NhanVien']]);
    return { success: true };
  } catch (e) {
    return { success: false, error: e.toString() };
  } finally { lock.releaseLock(); }
}

function deleteEmployeeRecord(username) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch (e) { return { success: false, error: 'Hệ thống đang bận, thử lại sau.' }; }
  try {
    const sh = _empSheet();
    const lastRow = sh.getLastRow();
    if (lastRow < 2) return { success: false, error: 'Không có nhân viên nào.' };
    const rows = sh.getRange(2, 1, lastRow - 1, 4).getValues();
    const clean = (username || '').toString().trim();
    let targetIdx = -1;
    for (let i = 0; i < rows.length; i++) {
      if (rows[i][1].toString().trim() === clean) { targetIdx = i; break; }
    }
    if (targetIdx === -1) return { success: false, error: 'Không tìm thấy nhân viên này.' };
    const role = (rows[targetIdx][3] || 'NhanVien').toString();
    if (role === 'Admin') {
      const adminCount = rows.filter(r => (r[3] || '') === 'Admin').length;
      if (adminCount <= 1) return { success: false, error: 'Không thể xóa Admin duy nhất còn lại.' };
    }
    sh.deleteRow(targetIdx + 2);
    return { success: true };
  } catch (e) {
    return { success: false, error: e.toString() };
  } finally { lock.releaseLock(); }
}

      zaloChatId: (r[4] || '').toString().trim()
    }));
  } catch (e) { return []; }
}

function _ensureMasterProducts(sh) {
  // [FIX] Khóa để 2 request chạy song song không cùng ghi KTC/S2... gây trùng dòng
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch (e) { return; }
  try {
    const lastRow = sh.getLastRow();
    const existingIds = {};
    if (lastRow > 1) {
      sh.getRange(2, 1, lastRow - 1, 1).getValues().forEach(r => {
        if (r[0] !== '' && r[0] !== null) existingIds[r[0].toString().trim().toUpperCase()] = true;
      });
    }
    const deletedIds = _getDeletedMasterIds_();
    const missing = MASTER_PRODUCTS.filter(p => !existingIds[p.id.toUpperCase()] && deletedIds.indexOf(p.id.toUpperCase()) === -1);
    if (!missing.length) return;
    const rows = missing.map(p => [
      p.id, p.name, p.img, p.normal, p.si_menu, p.shopee, p.si_cua_hang, p.si_khanh_quyen,
      p.si_200_400, p.si_400_600, p.si_600, p.si_hoi_cho, p.si_hoi_cho_TM, p.detailName || '', p.category || ''
    ]);
    sh.getRange(sh.getLastRow() + 1, 1, rows.length, 15).setValues(rows);
  } finally { lock.releaseLock(); }
}

function _getDeletedMasterIds_() {
  try { return JSON.parse(PropertiesService.getScriptProperties().getProperty('deletedMasterIds') || '[]'); } catch (e) { return []; }
}
function _markMasterDeleted_(id) {
  try {
    const key = (id || '').toString().toUpperCase();
    const list = _getDeletedMasterIds_();
    if (key && list.indexOf(key) === -1) { list.push(key); PropertiesService.getScriptProperties().setProperty('deletedMasterIds', JSON.stringify(list)); }
  } catch (e) {}
}
function _unmarkMasterDeleted_(id) {
  try {
    const key = (id || '').toString().toUpperCase();
    const list = _getDeletedMasterIds_();
    const idx = list.indexOf(key);
    if (idx !== -1) { list.splice(idx, 1); PropertiesService.getScriptProperties().setProperty('deletedMasterIds', JSON.stringify(list)); }
  } catch (e) {}
}

function getProductsCached() {
  try {
    const cache = CacheService.getScriptCache();
    const cached = cache.get(PRODUCT_CACHE_KEY);
    if (cached) return JSON.parse(cached);
  } catch (e) {}
  const list = getNewProductsFromSheet();
  try {
    CacheService.getScriptCache().put(PRODUCT_CACHE_KEY, JSON.stringify(list), 300);
  } catch (e) {}
  return list;
}
function _invalidateProductCache() {
  try { CacheService.getScriptCache().remove(PRODUCT_CACHE_KEY); } catch (e) {}
}

// Danh sách danh mục hiện có (mặc định + phát sinh từ dữ liệu)
function getProductCategories() {
  try {
    const list = getProductsCached();
    const set = {};
    DEFAULT_CATEGORIES.forEach(c => set[c] = 1);
    list.forEach(p => { if (p.category) set[p.category] = 1; });
    return Object.keys(set);
  } catch (e) { return DEFAULT_CATEGORIES.slice(); }
}

function saveProductToSheet(p) {
  try {
    const sh = _sheet('SanPham', SP_HEADERS);
    const lastRow = sh.getLastRow();
    const ids = lastRow > 1 ? sh.getRange(2, 1, lastRow - 1, 1).getValues() : [];
    _unmarkMasterDeleted_(p.id);
    const row = [
      p.id.toUpperCase(), p.name, p.img,
      Number(p.normal), Number(p.si_menu), Number(p.shopee),
      Number(p.si_cua_hang), Number(p.si_khanh_quyen), Number(p.si_200_400),
      Number(p.si_400_600), Number(p.si_600), Number(p.si_hoi_cho), Number(p.si_hoi_cho_TM),
      p.detailName || '', p.category || ''
    ];
    for (let i = 0; i < ids.length; i++) {
      if (ids[i][0].toString().toUpperCase() === p.id.toString().toUpperCase()) {
        sh.getRange(i + 2, 1).setNumberFormat('@'); // [FIX] giữ số 0 đầu nếu Mã SP toàn số
        sh.getRange(i + 2, 1, 1, row.length).setValues([row]);
        _invalidateProductCache();
        return 'Cập nhật sản phẩm thành công!';
      }
    }
    sh.getRange(sh.getLastRow() + 1, 1).setNumberFormat('@');
    sh.appendRow(row);
    _invalidateProductCache();
    return 'Thêm sản phẩm mới thành công!';
  } catch (e) { return 'Lỗi xử lý sản phẩm: ' + e.toString(); }
}

// Chạy tay 1 lần trong Apps Script editor để dọn các dòng trùng mã đang có sẵn (giữ dòng đầu tiên)
function cleanDuplicateProducts() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('SanPham');
  if (!sh || sh.getLastRow() < 2) return 'Không có dữ liệu.';
  const ids = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues();
  const firstIdx = {};
  ids.forEach((r, i) => {
    const k = r[0].toString().trim().toUpperCase();
    if (k && firstIdx[k] === undefined) firstIdx[k] = i;
  });
  let removed = 0;
  for (let i = ids.length - 1; i >= 0; i--) {   // duyệt từ dưới lên
    const k = ids[i][0].toString().trim().toUpperCase();
    if (k && firstIdx[k] !== i) { sh.deleteRow(i + 2); removed++; }
  }
  _invalidateProductCache();
  return 'Đã xóa ' + removed + ' dòng trùng mã.';
}

function _sanitizeNote(note) {
  if (!note) return '';
  const s = note.toString();
  const looksLikeReceiptEcho =
    /^\s*Ngày:\s*\d{1,2}\/\d{1,2}\/\d{4}[\s\S]*Tên:/i.test(s) ||
    s.indexOf('\nSản phẩm:\n') !== -1 || s.indexOf('Sản phẩm:\n') === 0;
  if (looksLikeReceiptEcho) return '';
  return s.length > 500 ? s.slice(0, 500) + ' …(đã rút gọn)' : s;
}

function _appendOrderRows(sh, data) {
  const n = data.cartDetails.length;
  if (!n) return true;

  let timestamp = new Date();
  if (data.orderDate) {
    const d = new Date(data.orderDate);
    if (!isNaN(d.getTime())) timestamp = d;
  }
  const empName = data.employeeName ? data.employeeName.toString().trim() : '';
  const phoneStr = (data.phone || '').toString().trim();
  const noteClean = _sanitizeNote(data.note);
  const startRow = sh.getLastRow() + 1;

  const blockAK = data.cartDetails.map((item, index) => {
    const row = new Array(11).fill('');
    if (index === 0) {
      row[0] = timestamp; row[1] = data.name; row[2] = phoneStr; row[3] = data.address;
      row[7] = data.priceType; row[8] = noteClean; row[9] = data.taxCode; row[10] = data.company;
    }
    row[4] = item.name + ' (' + item.id + ')';
    row[5] = item.qty;
    row[6] = item.price * item.qty;
    return row;
  });

  const blockO = data.cartDetails.map(item => [(item.detailName && item.detailName.trim()) || item.name]);
  const blockS = data.cartDetails.map((item, index) => [index === 0 ? empName : '']);
  const blockInvoice = data.cartDetails.map((item, index) => [index === 0 ? (data.invoiceName || '') : '']);

  // [FIX] ép VĂN BẢN mọi cột chữ: B tên, C SĐT, D địa chỉ, H loại giá, I ghi chú, J MST, K công ty, O tên chi tiết, S nhân viên, W tên hóa đơn.
  // Trước đây chỉ ép C và J nên ghi chú như "1/2", "3-4" bị Sheets đổi thành NGÀY (làm trang Lịch Sử trả rỗng).
  sh.getRangeList([2, 3, 4, 8, 9, 10, 11, 15, 19, 23].map(function (c) { return _a1Col_(c) + startRow + ':' + _a1Col_(c) + (startRow + n - 1); })).setNumberFormat('@');

  sh.getRange(startRow, 19, n, 1).setValues(blockS);
  sh.getRange(startRow, 23, n, 1).setValues(blockInvoice);   // ← thêm dòng này

  return true;
}

  const lock = LockService.getScriptLock();

    if (!totals) return { success: false, error: 'Không đọc được dữ liệu đơn hàng.' };

    for (let i = 0; i < data.length; i++) {
      const r = data[i];

      if (!name || _looksLikeUrl(name)) continue;
      const codeKey = _stripDiacriticsServer(code).replace(/\s+/g, '');

      const hay = _stripDiacriticsServer([
        name, code, r[DKS_COL.OLD_CODE - 1], r[DKS_COL.DESCRIPTION - 1], r[DKS_COL.PHONE - 1], r[DKS_COL.ADDRESS - 1],
        r[DKS_COL.INVOICE_NAME - 1], r[DKS_COL.COMPANY - 1], r[DKS_COL.TAX - 1], r[DKS_COL.CCCD - 1]
      ].join(' '));

      if (!ok) continue;
      const rank = codeKey.indexOf(qNoSpace) === 0 ? 0 : (codeKey.indexOf(qNoSpace) > 0 ? 1 : (nameKey.indexOf(q) === 0 ? 2 : 3));
      found.push({
        rank: rank, code: code,

          phone: (r[DKS_COL.PHONE - 1] || '').toString(),

// Thứ tự nhận diện khách: (1) theo Mã KH, (2) theo SĐT. Khách import từ file mới có thể chưa có SĐT -> nhận theo mã.
// extra = { customerCode, codeGroup, codeFactory, cccd }
function _saveNewCustomerInternal(name, phone, address, customerType, city, taxCode, companyName, invoiceName, extra) {
  extra = extra || {};
  const sh = _dataKhachSiSheet();
  if (!sh) return '';
  const pc = (phone || '').toString().replace(/[^0-9]/g, '');
  const inCode = _normCode_(extra.customerCode);
  if (!pc && !inCode) return '';
  const today = new Date();
  const lastRow = sh.getLastRow();
  const WIDTH = DKS_TOTAL_COLS;
  // [TỐC ĐỘ] chỉ đọc 2 cột Mã + SĐT để tìm khách (không đọc cả bảng); khi cần cập nhật mới đọc đúng 1 dòng
  const nRows = lastRow > 1 ? lastRow - 1 : 0;
  const codeCol = nRows ? sh.getRange(2, DKS_COL.CODE, nRows, 1).getValues() : [];
  const phoneCol = nRows ? sh.getRange(2, DKS_COL.PHONE, nRows, 1).getValues() : [];

  let idx = -1;
  if (inCode) {
    for (let i = 0; i < codeCol.length; i++) {
      if (_normCode_(codeCol[i][0]) === inCode) { idx = i; break; }
    }
  }
  if (idx < 0 && pc) {
    for (let i = 0; i < phoneCol.length; i++) {
      const rowPhone = (phoneCol[i][0] || '').toString().replace(/[^0-9]/g, '');
      if (rowPhone === pc) { idx = i; break; }
    }
  }
  if (idx < 0 && !pc) return '';   // có mã nhưng mã không còn trong sheet và không có SĐT -> không tạo dòng rỗng

  const allCodes = codeCol.map(r => r[0]);

  if (idx >= 0) {
    const row = sh.getRange(idx + 2, 1, 1, WIDTH).getValues()[0];
    row[DKS_COL.NAME - 1] = name;
    row[DKS_COL.ADDRESS - 1] = address;
    row[DKS_COL.LAST_ORDER - 1] = today;
    if (!row[DKS_COL.FIRST_ORDER - 1]) row[DKS_COL.FIRST_ORDER - 1] = today;
    if (pc && !(row[DKS_COL.PHONE - 1] || '').toString().replace(/[^0-9]/g, '')) row[DKS_COL.PHONE - 1] = pc;
    if (customerType) row[DKS_COL.LOAI_KHACH_SI - 1] = customerType;
    if (taxCode) row[DKS_COL.TAX - 1] = taxCode;
    if (companyName) row[DKS_COL.COMPANY - 1] = companyName;
    if (invoiceName) row[DKS_COL.INVOICE_NAME - 1] = invoiceName;
    if (extra.cccd) row[DKS_COL.CCCD - 1] = extra.cccd;
    // Khách cũ chưa có mã -> cấp mã mới theo quy định
    if (!_normCode_(row[DKS_COL.CODE - 1]) && extra.codeGroup) {
      const gen = _generateCode_(allCodes, extra.codeGroup, extra.codeFactory, city, address);
      if (gen.code) {
        row[DKS_COL.CODE - 1] = gen.code; row[DKS_COL.FACTORY - 1] = gen.factory;
        row[DKS_COL.PROV - 1] = gen.prov; row[DKS_COL.GROUP - 1] = gen.group;
      }
    }
    const r1 = idx + 2;
    sh.getRangeList([_a1Col_(DKS_COL.PHONE) + r1, _a1Col_(DKS_COL.TAX) + r1, _a1Col_(DKS_COL.CCCD) + r1, _a1Col_(DKS_COL.CODE) + r1]).setNumberFormat('@');
    sh.getRange(r1, 1, 1, WIDTH).setValues([row]);
    return (row[DKS_COL.CODE - 1] || '').toString();
  }

  const newRow = sh.getLastRow() + 1;
  const arr = new Array(WIDTH).fill('');
  arr[DKS_COL.NAME - 1] = name;
  arr[DKS_COL.PHONE - 1] = pc;
  arr[DKS_COL.ADDRESS - 1] = address;
  arr[DKS_COL.FIRST_ORDER - 1] = today;
  arr[DKS_COL.LAST_ORDER - 1] = today;
  if (customerType) arr[DKS_COL.LOAI_KHACH_SI - 1] = customerType;
  if (taxCode) arr[DKS_COL.TAX - 1] = taxCode;
  if (companyName) arr[DKS_COL.COMPANY - 1] = companyName;
  if (invoiceName) arr[DKS_COL.INVOICE_NAME - 1] = invoiceName;
  if (extra.cccd) arr[DKS_COL.CCCD - 1] = extra.cccd;
  if (extra.codeGroup) {
    const gen = _generateCode_(allCodes, extra.codeGroup, extra.codeFactory, city, address);
    if (gen.code) {
      arr[DKS_COL.CODE - 1] = gen.code; arr[DKS_COL.FACTORY - 1] = gen.factory;
      arr[DKS_COL.PROV - 1] = gen.prov; arr[DKS_COL.GROUP - 1] = gen.group;
    }
  }
  sh.getRangeList([_a1Col_(DKS_COL.PHONE) + newRow, _a1Col_(DKS_COL.TAX) + newRow, _a1Col_(DKS_COL.CCCD) + newRow, _a1Col_(DKS_COL.CODE) + newRow]).setNumberFormat('@');
  sh.getRange(newRow, 1, 1, WIDTH).setValues([arr]);
  return (arr[DKS_COL.CODE - 1] || '').toString();
}

const _READ_CHUNK = 1000;

// countedOnly = true: chỉ lấy các đơn ĐƯỢC TÍNH doanh thu/KPI (đơn trước 01/10/2026, hoặc đơn từ 01/10/2026 đã được duyệt).
// countedOnly rỗng/false: lấy toàn bộ đơn như cũ (lịch sử, tra cứu, sao chép...).
function _readDonHangGrouped(fromDate, toDate, empFilter, countedOnly) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('DonHang');
  if (!sh || sh.getLastRow() < 2) return [];
  const lastRow = sh.getLastRow();

  const from = fromDate ? new Date(fromDate) : null;
  const to = toDate ? new Date(toDate) : null;
  if (to) to.setHours(23, 59, 59, 999);
  const empSet = (empFilter && empFilter.length) ? empFilter.map(e => (e || '').toString().toLowerCase()) : null;

  let data;
  if (false) {   // [SỬA] tắt dừng-quét-sớm: nó bỏ sót đơn khi các dòng không xếp đúng thứ tự ngày; đọc toàn bộ cho chắc
    let collected = [];
    let cursor = lastRow;
    let doneScanning = false;
    while (cursor >= 2 && !doneScanning) {
      const chunkStart = Math.max(2, cursor - _READ_CHUNK + 1);
      const chunk = sh.getRange(chunkStart, 1, cursor - chunkStart + 1, READ_WIDTH).getValues();
      collected = chunk.concat(collected);
      for (let i = 0; i < chunk.length; i++) {
        if (chunk[i][0] && chunk[i][0] !== '') {
          const d = chunk[i][0] instanceof Date ? chunk[i][0] : new Date(chunk[i][0]);
          if (d < from) { doneScanning = true; break; }
        }
      }
      cursor = chunkStart - 1;
    }
    data = collected;
  } else {
    data = sh.getRange(2, 1, lastRow - 1, READ_WIDTH).getValues();
  }

  let startRowOfData = lastRow - data.length + 1;
  // [LƯU TRỮ] bổ sung tháng cũ đã lưu trữ (chỉ khi tháng đó KHÔNG còn ở sheet chính => không bao giờ đếm trùng)
  let _archN = 0;
  try {
    if (typeof _ldExtraRows_ === 'function') {
      const ex = _ldExtraRows_(from, to);
      if (ex.rows.length) { data = ex.rows.concat(data); _archN = ex.rows.length; }
    }
  } catch (eA) { Logger.log('Lỗi đọc lưu trữ: ' + eA); throw eA; }   // lỗi thì báo rõ, không âm thầm đọc thiếu

  const orders = [];
  let cur = null, curSkip = false;
  data.forEach((r, idx) => {
    if (r[0] && r[0] !== '') {
      curSkip = (r[COL_STATUS - 1] || '') === 'Huy';
      if (!curSkip) {
        const d = r[0] instanceof Date ? r[0] : new Date(r[0]);
        if ((from && d < from) || (to && d > to)) { curSkip = true; cur = null; return; }
        const emp = (r[18] || '').toString();
        if (empSet && !empSet.includes(emp.toLowerCase())) { curSkip = true; cur = null; return; }
        const approvedByRaw = (r[23] || '').toString().trim();
        const needsAppr = _orderNeedsApproval(d);
        const apprProblem = (needsAppr && approvedByRaw) ? _approvalProblem_(approvedByRaw, r[24]) : '';
        const approvedBy = apprProblem ? '' : approvedByRaw;   // duyệt không hợp lệ = CHƯA duyệt (web sẽ hiện "Chờ duyệt" + nút Duyệt)
        const approvedAt = apprProblem ? '' : (r[24] instanceof Date ? r[24].toISOString() : (r[24] || ''));
        const counted = !needsAppr || !!approvedBy;
        if (countedOnly && !counted) { curSkip = true; cur = null; return; }
        cur = {
          date: d.toISOString(), name: _hsv_(r[1] || ''), phone: _hsv_(r[2] || ''), address: _hsv_(r[3] || ''),
          priceType: _hsv_(r[7] || ''), emp: emp, note: _hsv_(r[8] || ''), taxCode: _hsv_(r[9] || ''), company: _hsv_(r[10] || ''),
          invoiceName: _hsv_(r[22] || ''),
          approvedBy: approvedBy, approvedAt: approvedAt, needsApproval: needsAppr, counted: counted, approvalProblem: apprProblem,   // ← thêm dòng này (cột 23, index 22)
          items: [], total: 0, sheetStartRow: idx < _archN ? -(idx + 1) : startRowOfData + (idx - _archN), archived: idx < _archN
        };
        orders.push(cur);
        // KHÔNG return ở đây nữa - để dòng này rơi xuống xử lý sản phẩm bên dưới
      } else {
        cur = null;
        return;
      }
    }
    if (curSkip || !cur) return;
    const val = Number(r[6]) || 0, qty = Number(r[5]) || 0;
    const rawCell = (r[4] || '').toString();               // dạng "Tên SP (MÃ)" luôn cố định, không đổi
    const idMatch = rawCell.match(/\(([^)]+)\)\s*$/);
    const prodId = idMatch ? idMatch[1] : '';               // [MỚI] lấy đúng Mã SP thật
    const prodName = r[14] ? r[14].toString() : rawCell;
    cur.total += val;
    cur.items.push({ name: prodName, qty: qty, price: val, unitPrice: qty ? Math.round(val / qty) : 0, id: prodId });
  });

  return orders;
}

function getHistoryPage(fromDate, toDate, empFilter, searchStr, countedOnly) {
  try {
    let empArr = null;
    if (empFilter) {
      empArr = Array.isArray(empFilter) ? empFilter.filter(Boolean) : [empFilter];
      if (!empArr.length) empArr = null;
    }
    let orders = _readDonHangGrouped(fromDate, toDate, empArr, !!countedOnly);
    if (searchStr) {
      const s = searchStr.toString().toLowerCase();
      orders = orders.filter(o => (o.name || '').toLowerCase().indexOf(s) !== -1 || (o.phone || '').toString().indexOf(s) !== -1);
    }
    const totalRev = orders.reduce((a, o) => a + o.total, 0);
    return { orders: orders, totalRev: totalRev, count: orders.length };
  } catch (e) {
    return { orders: [], totalRev: 0, count: 0, error: e.toString() };
  }
}

function getRevenueData(fromDate, toDate, employees) {
  try {
    const allOrders = _readDonHangGrouped(fromDate, toDate, null, true);
    const empMap = {};
    let systemTotal = 0;
    allOrders.forEach(o => {
      systemTotal += o.total;
      if (!empMap[o.emp]) empMap[o.emp] = { orders: 0, qty: 0, total: 0 };
      empMap[o.emp].orders++;
      empMap[o.emp].total += o.total;
      empMap[o.emp].qty += o.items.reduce((a, it) => a + it.qty, 0);
    });
    const keys = (employees && employees.length) ? employees : Object.keys(empMap);
    let filtTotal = 0, filtOrders = 0, filtQty = 0;
    const table = keys.filter(k => empMap[k]).map(name => {
      const d = empMap[name];
      filtTotal += d.total; filtOrders += d.orders; filtQty += d.qty;
      return { name: name, orders: d.orders, total: d.total, qty: d.qty, pct: systemTotal ? (d.total / systemTotal * 100) : 0 };
    });
    return { table: table, total: filtTotal, orders: filtOrders, qty: filtQty, systemTotal: systemTotal };
  } catch (e) {
    return { table: [], total: 0, orders: 0, qty: 0, systemTotal: 0, error: e.toString() };
  }
}

function getAnalytics(fromDate, toDate) {
  try {
    const orders = _readDonHangGrouped(fromDate, toDate, null, true);
    const byProduct = {};
    const byDay = {};
    const byPriceType = {};
    let totalRev = 0, totalQty = 0;

    orders.forEach(o => {
      totalRev += o.total;
      const day = Utilities.formatDate(new Date(o.date), Session.getScriptTimeZone(), 'dd/MM');
      if (!byDay[day]) byDay[day] = { orders: 0, total: 0 };
      byDay[day].orders++; byDay[day].total += o.total;

      if (!byPriceType[o.priceType]) byPriceType[o.priceType] = { orders: 0, total: 0 };
      byPriceType[o.priceType].orders++; byPriceType[o.priceType].total += o.total;

      o.items.forEach(it => {
        totalQty += it.qty;
        if (!byProduct[it.name]) byProduct[it.name] = { qty: 0, total: 0 };
        byProduct[it.name].qty += it.qty; byProduct[it.name].total += it.price;
      });
    });

    const topProducts = Object.keys(byProduct).map(name => ({ name: name, qty: byProduct[name].qty, total: byProduct[name].total }))
      .sort((a, b) => b.total - a.total).slice(0, 15);
    const dayStats = Object.keys(byDay).map(d => ({ day: d, orders: byDay[d].orders, total: byDay[d].total }));
    const priceTypeStats = Object.keys(byPriceType).map(k => ({ priceType: k, orders: byPriceType[k].orders, total: byPriceType[k].total }))
      .sort((a, b) => b.total - a.total);

    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('DonHang');
    let newCust = 0, returningCust = 0;
    if (sh && sh.getLastRow() > 1) {
      let all = sh.getRange(2, 1, sh.getLastRow() - 1, READ_WIDTH).getValues();
      try { if (typeof _ldExtraRows_ === 'function') all = _ldExtraRows_(null, null).rows.concat(all); } catch (eA) { Logger.log('Lỗi đọc lưu trữ: ' + eA); }
      const firstBuy = {};
      all.forEach(r => {
        if (r[0] && r[0] !== '' && (r[COL_STATUS - 1] || '') !== 'Huy') {
          const ph = (r[2] || '').toString().replace(/[^0-9]/g, '');
          const d = r[0] instanceof Date ? r[0] : new Date(r[0]);
          if (ph && (!firstBuy[ph] || d < firstBuy[ph])) firstBuy[ph] = d;
        }
      });
      const from = fromDate ? new Date(fromDate) : null;
      const seenInRange = {};
      orders.forEach(o => {
        const ph = (o.phone || '').toString().replace(/[^0-9]/g, '');
        if (!ph || seenInRange[ph]) return;
        seenInRange[ph] = 1;
        if (from && firstBuy[ph] && firstBuy[ph] >= from) newCust++; else returningCust++;
      });
    }

    return {
      totalRev: totalRev, totalOrders: orders.length, totalQty: totalQty,
      aov: orders.length ? Math.round(totalRev / orders.length) : 0,
      topProducts: topProducts, dayStats: dayStats, priceTypeStats: priceTypeStats,
      newCust: newCust, returningCust: returningCust
    };
  } catch (e) {
    return { error: e.toString() };
  }
}

// ════════════════════════════════════════════════════════════
//  BÁO CÁO TỔNG HỢP: theo khách sỉ, theo nhân viên, so sánh doanh thu
//  theo tháng (kèm % so với TB các tháng có dữ liệu)
// ════════════════════════════════════════════════════════════
function getSummaryReport(fromDate, toDate) {
  try {
    const orders = _readDonHangGrouped(fromDate, toDate, null, true);

    // Theo khách sỉ (gộp theo SĐT)
    const byCust = {};
    orders.forEach(o => {
      const key = (o.phone || o.name || 'Khách lẻ').toString();
      if (!byCust[key]) byCust[key] = { name: o.name, phone: o.phone, orders: 0, qty: 0, revenue: 0, products: {} };
      const c = byCust[key];
      c.orders++;
      c.revenue += o.total;
      o.items.forEach(it => {
        c.qty += it.qty;
        if (!c.products[it.name]) c.products[it.name] = { qty: 0, revenue: 0 };
        c.products[it.name].qty += it.qty;
        c.products[it.name].revenue += it.price;
      });
    });
    const byCustomer = Object.keys(byCust).map(k => {
      const c = byCust[k];
      return {
        name: c.name, phone: c.phone, orders: c.orders, qty: c.qty, revenue: c.revenue,
        products: Object.keys(c.products).map(pn => ({ name: pn, qty: c.products[pn].qty, revenue: c.products[pn].revenue }))
          .sort((a, b) => b.revenue - a.revenue)
      };
    }).sort((a, b) => b.revenue - a.revenue);

    // Theo nhân viên - [SỬA] bỏ qua đơn không có tên NV (không hiện dòng trống)
    const byEmpMap = {};
    orders.forEach(o => {
      if (!o.emp || !o.emp.trim()) return;
      if (!byEmpMap[o.emp]) byEmpMap[o.emp] = { orders: 0, qty: 0, revenue: 0 };
      byEmpMap[o.emp].orders++;
      byEmpMap[o.emp].revenue += o.total;
      byEmpMap[o.emp].qty += o.items.reduce((a, it) => a + it.qty, 0);
    });
    const byEmployee = Object.keys(byEmpMap).map(k => ({ name: k, orders: byEmpMap[k].orders, qty: byEmpMap[k].qty, revenue: byEmpMap[k].revenue }))
      .sort((a, b) => b.revenue - a.revenue);

    // So sánh theo tháng (dùng đúng khoảng orders đã đọc, nhóm theo yyyy-MM)
    const byMonthMap = {};
    orders.forEach(o => {
      const d = new Date(o.date);
      const key = d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2);
      if (!byMonthMap[key]) byMonthMap[key] = { revenue: 0, orders: 0 };
      byMonthMap[key].revenue += o.total;
      byMonthMap[key].orders++;
    });
    const monthKeys = Object.keys(byMonthMap).sort();
    const avgRevenue = monthKeys.length ? monthKeys.reduce((a, k) => a + byMonthMap[k].revenue, 0) / monthKeys.length : 0;
    const monthlyComparison = monthKeys.map(k => {
      const rev = byMonthMap[k].revenue;
      const pctVsAvg = avgRevenue ? ((rev - avgRevenue) / avgRevenue * 100) : 0;
      return { month: k, revenue: rev, orders: byMonthMap[k].orders, pctVsAvg: pctVsAvg };
    });

    return { byCustomer: byCustomer, byEmployee: byEmployee, monthlyComparison: monthlyComparison, avgRevenue: avgRevenue };
  } catch (e) {
    return { error: e.toString(), byCustomer: [], byEmployee: [], monthlyComparison: [], avgRevenue: 0 };
  }
}

    const now = new Date();
    return Object.entries(lastBuy).map(([phone, info]) => {
      const days = Math.floor((now - info.date) / (1000 * 60 * 60 * 24));
      return { phone: phone, name: info.name, lastDate: info.date.toISOString(), daysSince: days };
    }).filter(x => x.daysSince >= threshold).sort((a, b) => b.daysSince - a.daysSince);
  } catch (e) { return []; }
}

// hạn (giá Hội Chợ + (>3 ngày hoặc >3 đơn) + còn nợ).
function getCongNoCustomerList() {
  try {
    const sh = _congNoSheet();
    const data = sh.getDataRange().getValues();
    if (data.length <= 1) return [];
    const now = new Date();
    const byCust = {};
    data.slice(1).forEach((r, i) => {
      const sheetRow = i + 2;
      const name = (r[1] || '').toString();
      const phone = (r[2] || '').toString();
      const key = phone || name || ('row' + sheetRow);
      const ngay = r[3] instanceof Date ? r[3] : new Date(r[3]);
      const total = Number(r[4]) || 0;
      const conLai = r[12] !== '' && r[12] !== null && r[12] !== undefined ? Number(r[12]) : total;
      const loaiGia = (r[14] || '').toString();
      if (!byCust[key]) byCust[key] = { name: name, phone: phone, orders: 0, unpaidOrders: 0, total: 0, conLai: 0, oldestUnpaidDays: 0, warning: false };
      const c = byCust[key];
      c.name = name || c.name; c.phone = phone || c.phone;
      c.orders++;
      c.total += total;
      c.conLai += conLai;
      if (conLai > 0) {
        c.unpaidOrders++;
        const days = Math.floor((now - ngay) / (1000 * 60 * 60 * 24));
        if (days > c.oldestUnpaidDays) c.oldestUnpaidDays = days;
        const isHoiCho = /hoi.?cho/i.test(loaiGia) || loaiGia.indexOf('Hội Chợ') !== -1;
        if (isHoiCho && (days > 3 || c.unpaidOrders > 3)) c.warning = true;
      }
    });
    return Object.keys(byCust).map(k => byCust[k]).sort((a, b) => b.conLai - a.conLai);
  } catch (e) { return []; }
}

    if (data.length <= 1) return [];
    const now = new Date();

    data.slice(1).forEach((r, i) => {
      const sheetRow = i + 2;
      const name = (r[1] || '').toString();
      const phone = (r[2] || '').toString();
      if (phone !== key && name !== key) return;
      const ngay = r[3] instanceof Date ? r[3] : new Date(r[3]);
      const total = Number(r[4]) || 0;
      const daThu = r[11] !== '' && r[11] !== null && r[11] !== undefined ? Number(r[11]) : 0;
      const conLai = r[12] !== '' && r[12] !== null && r[12] !== undefined ? Number(r[12]) : (total - daThu);
      let lichSu = [];
      try { lichSu = JSON.parse(r[13] || '[]'); } catch (e) {}

        sheetRow: sheetRow, date: ngay.toISOString(), name: name, phone: phone,
        total: total, paid: daThu, remaining: conLai, history: lichSu,
        status: r[6] || '', priceType: loaiGia, daysSince: daysSince, emp: r[5] || ''
      });
    });

    });

}

function recordDebtPayment(sheetRow, amount, payDate, confirmedBy) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch (e) { return { success: false, error: 'Hệ thống đang bận, thử lại sau.' }; }
  try {
    const sh = _congNoSheet();
    const r = sh.getRange(sheetRow, 1, 1, 15).getValues()[0];
    const total = Number(r[4]) || 0;
    let daThu = Number(r[11]) || 0;
    let lichSu = [];
    try { lichSu = JSON.parse(r[13] || '[]'); } catch (e) {}
    const amt = Math.max(0, Number(amount) || 0);
    if (amt <= 0) return { success: false, error: 'Số tiền không hợp lệ.' };
    lichSu.push({ date: (payDate ? new Date(payDate) : new Date()).toISOString(), amount: amt, by: confirmedBy || '' });
    daThu += amt;
    if (daThu > total) daThu = total;
    const conLai = Math.max(0, total - daThu);
    const status = conLai <= 0 ? 'Hoan tat' : 'Dang thu mot phan';
    sh.getRange(sheetRow, 12).setValue(daThu);
    sh.getRange(sheetRow, 13).setValue(conLai);
    sh.getRange(sheetRow, 14).setValue(JSON.stringify(lichSu));
    sh.getRange(sheetRow, 7).setValue(status);
    if (conLai <= 0) { sh.getRange(sheetRow, 8).setValue(new Date()); sh.getRange(sheetRow, 9).setValue(confirmedBy || ''); }
    return { success: true, paid: daThu, remaining: conLai, status: status };
  } catch (e) {
    return { success: false, error: e.toString() };
  } finally { lock.releaseLock(); }
}

function exportMultiSheetExcel(sheetsSpec) {
  let tempSS = null;
  try {
    tempSS = SpreadsheetApp.create('TEMP_EXPORT_9SACH_' + new Date().getTime());
    const firstDefaultSheet = tempSS.getSheets()[0];
    const usedNames = {};
    (sheetsSpec || []).forEach((spec, idx) => {
      const sh = idx === 0 ? firstDefaultSheet : tempSS.insertSheet();
      // [SỬA] tên sheet: bỏ ký tự cấm [ ] : * ? / \ , cắt 90 ký tự, không trùng
      let nm = (spec.title || ('Sheet' + (idx + 1))).toString().replace(/[\[\]:*?\/\\]/g, ' ').trim().substring(0, 90) || ('Sheet' + (idx + 1));
      let base = nm, k = 2;
      while (usedNames[nm.toLowerCase()]) { nm = base.substring(0, 85) + ' ' + (k++); }
      usedNames[nm.toLowerCase()] = true;
      sh.setName(nm);
      const headers = spec.headers || [];
      // [SỬA] đồng bộ độ rộng mọi dòng (setValues báo lỗi nếu dòng dài/ngắn khác nhau), bỏ null/undefined
      // [FIX] không dùng Math.max(...mảng) - lỗi tràn stack khi >~100k dòng
      let width = Math.max(headers.length, 1);
      (spec.rows || []).forEach(r => { const l = (r || []).length; if (l > width) width = l; });
      const rows = (spec.rows || []).map(r => {
        const out = new Array(width).fill('');
        (r || []).forEach((v, i) => { out[i] = (v === null || v === undefined) ? '' : v; });
        return out;
      });
      // [FIX] sheet mới chỉ có 1000 dòng x 26 cột -> mở rộng trước khi ghi, nếu không báo lỗi "out of bounds"
      const needRows = rows.length + 1;
      if (sh.getMaxRows() < needRows) sh.insertRowsAfter(sh.getMaxRows(), needRows - sh.getMaxRows());
      if (sh.getMaxColumns() < width) sh.insertColumnsAfter(sh.getMaxColumns(), width - sh.getMaxColumns());
      if (headers.length) sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#1b4332').setFontColor('#ffffff');
      for (let st = 0; st < rows.length; st += 5000) {
        const part = rows.slice(st, st + 5000);
        sh.getRange(2 + st, 1, part.length, width).setValues(part);
      }
      if (width && rows.length <= 5000) sh.autoResizeColumns(1, width);
    });
    SpreadsheetApp.flush();
    const url = 'https://docs.google.com/spreadsheets/d/' + tempSS.getId() + '/export?format=xlsx';
    const resp = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, muteHttpExceptions: true });
    if (resp.getResponseCode() !== 200) throw new Error('Google trả về HTTP ' + resp.getResponseCode() + ' khi xuất file (kiểm tra quyền Drive/Sheets của script).');
    const base64 = Utilities.base64Encode(resp.getBlob().getBytes());
    DriveApp.getFileById(tempSS.getId()).setTrashed(true);
    return base64;
  } catch (e) {
    if (tempSS) { try { DriveApp.getFileById(tempSS.getId()).setTrashed(true); } catch (ex) {} }
    throw new Error('Lỗi xuất Excel: ' + e.toString());
  }
}

function _monthKey(d) {
  d = (d instanceof Date) ? d : new Date(d);
  return d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2);
}

function _readOrderBlockTotals(sh, startRow, rowCount) {

    const block = sh.getRange(startRow, 1, n, 7).getValues(); // cột A-G

}

// Cộng/trừ 1 delta vào đúng dòng tháng trong TongHopThang (tạo dòng mới nếu chưa có)
function _tongHopApplyDelta(dateObj, revenueDelta, ordersDelta, qtyDelta, phone) {
  try {
    const sh = _tongHopSheet();
    const key = _monthKey(dateObj);
    const lastRow = sh.getLastRow();
    const data = lastRow > 1 ? sh.getRange(2, 1, lastRow - 1, 6).getValues() : [];
    const cleanPhone = (phone || '').toString().replace(/\D/g, '');

    for (let i = 0; i < data.length; i++) {
      if (data[i][0] === key) {
        const rowIdx = i + 2;
        let custList = [];
        try { custList = JSON.parse(data[i][4] || '[]'); } catch (e) {}
        if (cleanPhone && custList.indexOf(cleanPhone) === -1) custList.push(cleanPhone);
        const newRev = Math.max(0, (Number(data[i][1]) || 0) + revenueDelta);
        const newOrders = Math.max(0, (Number(data[i][2]) || 0) + ordersDelta);
        const newQty = Math.max(0, (Number(data[i][3]) || 0) + qtyDelta);
        sh.getRange(rowIdx, 2, 1, 5).setValues([[newRev, newOrders, newQty, JSON.stringify(custList), new Date()]]);
        return;
      }
    }
    // chưa có dòng của tháng này -> tạo mới
    const custList = cleanPhone ? [cleanPhone] : [];
    const newRowIdx = sh.getLastRow() + 1;
    sh.getRange(newRowIdx, 1).setNumberFormat('@'); // [FIX v6.3] ép văn bản TRƯỚC khi ghi
    sh.getRange(newRowIdx, 1, 1, 6).setValues([[key, Math.max(0, revenueDelta), Math.max(0, ordersDelta), Math.max(0, qtyDelta), JSON.stringify(custList), new Date()]]);
  } catch (e) {
    Logger.log('Lỗi cập nhật TongHopThang: ' + e.toString());
  }
}

// [CHẠY TAY 1 LẦN] Dựng lại toàn bộ TongHopThang từ dữ liệu DonHang hiện có
function rebuildTongHopThang() {
  if (typeof _ldRebuildGuard_ === 'function' && _ldRebuildGuard_()) throw new Error('Đã có tháng được lưu trữ và xóa khỏi DonHang. Dùng sanDoanhThuTatCa() thay cho chức năng này.');
  const src = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('DonHang');
  if (!src || src.getLastRow() < 2) { Logger.log('DonHang trống.'); return; }
  const data = src.getRange(2, 1, src.getLastRow() - 1, READ_WIDTH).getValues();
  const monthMap = {};
  let curDate = null, curPhone = '', curSkip = false;

  data.forEach(r => {
    if (r[0] && r[0] !== '') {
      curDate = r[0] instanceof Date ? r[0] : new Date(r[0]);
      curPhone = (r[2] || '').toString().replace(/\D/g, '');
      curSkip = (r[COL_STATUS - 1] || '') === 'Huy';
      if (!curSkip && _orderNeedsApproval(curDate) && !_approvalValid_(r[23], r[24])) curSkip = true; // đơn từ 01/10/2026 chưa duyệt: không tính
      if (!curSkip) {
        const key = _monthKey(curDate);
        if (!monthMap[key]) monthMap[key] = { revenue: 0, orders: 0, qty: 0, cust: {} };
        monthMap[key].orders++;
        if (curPhone) monthMap[key].cust[curPhone] = 1;
      }
    }
    if (curSkip || !curDate) return;
    const key = _monthKey(curDate);
    if (!monthMap[key]) monthMap[key] = { revenue: 0, orders: 0, qty: 0, cust: {} };
    monthMap[key].revenue += Number(r[6]) || 0;
    monthMap[key].qty += Number(r[5]) || 0;
  });

  const sh = _tongHopSheet();
  const lastRow = sh.getLastRow();
  if (lastRow > 1) sh.getRange(2, 1, lastRow - 1, 6).clearContent();

  const rows = Object.keys(monthMap).sort().map(key => {
    const m = monthMap[key];
    return [key, m.revenue, m.orders, m.qty, JSON.stringify(Object.keys(m.cust)), new Date()];
  });
  if (rows.length) {
    sh.getRange(2, 1, rows.length, 1).setNumberFormat('@'); // [FIX v6.3] ép văn bản TRƯỚC khi ghi
    sh.getRange(2, 1, rows.length, 6).setValues(rows);
  }
  Logger.log('Đã dựng lại TongHopThang: ' + rows.length + ' tháng, tổng doanh thu = ' +
    rows.reduce((a, r) => a + r[1], 0).toLocaleString('vi-VN') + 'đ');
}

function submitOrder(data, isDebt, editInfo) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(25000);
  } catch (e) {
    throw new Error('Hệ thống đang bận, vui lòng thử lại sau vài giây.');
  }
  try {
    // [MỚI] Chặn ĐƠN TRỐNG (không có sản phẩm)
    if (!data || !Array.isArray(data.cartDetails) || !data.cartDetails.some(it => Number(it.qty) > 0)) {
      return { success: false, empty: true, message: 'Đơn hàng trống - chưa có sản phẩm nào, không tạo đơn.' };
    }
    // [MỚI] Chặn ĐƠN TRÙNG: cùng NV + cùng khách + cùng sản phẩm/số lượng/tổng tiền trong vòng 3 phút
    let dupKey = '';
    if (!editInfo || !editInfo.startRow) {
      const sig = [data.employeeName || '', (data.phone || '').toString().replace(/[^0-9]/g, '') || data.name || '', Number(data.totalAmount) || 0,
        data.cartDetails.map(it => it.id + 'x' + it.qty).sort().join(',')].join('|');
      dupKey = 'ORD_' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, sig, Utilities.Charset.UTF_8));
      if (!data.forceDuplicate && CacheService.getScriptCache().get(dupKey)) {
        return { success: false, duplicate: true, message: 'Đơn này giống hệt một đơn vừa tạo cách đây chưa tới 3 phút (cùng khách, cùng sản phẩm, cùng tiền).' };
      }
    }
    const sh = _donHangSheet();
    // [GĐ4] đơn có ngày thuộc tháng đã lưu trữ + xóa nguồn: từ chối NGAY, trước khi hủy đơn cũ (khi sửa) hay ghi gì
    if (_wgMonthLocked_(data.orderDate || new Date())) return { success: false, message: WG_MSG_MONTH_LOCKED };
    let oldTotals = null;
    let oldCounted = true;

    if (editInfo && editInfo.startRow) {
      const chkEdit = _wgCheckBlock_(sh, editInfo.startRow, editInfo.rowCount, editInfo.fp);
      if (!chkEdit.ok) return { success: false, message: chkEdit.error };
      oldTotals = _readOrderBlockTotals(sh, editInfo.startRow, editInfo.rowCount);
      oldCounted = _isBlockCounted(sh, editInfo.startRow);
      const empName = (data.employeeName || '').toString().trim();
      const note = 'Đã SỬA lúc ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm dd/MM/yyyy') +
        (empName ? ' bởi ' + empName : '') + ' → thay bằng đơn mới bên dưới.';
      _cancelOrderRows(sh, editInfo.startRow, editInfo.rowCount, note);
    }

    // [MỚI] Khách CHƯA có mã: lưu vào sheet KhachMoiChoDuyet (chưa vào DataKhachSi) (không báo Zalo).
    // Làm TRƯỚC khi ghi đơn để lỗi cấp mã không để lại đơn mồ côi.
    let usedPending = false, pendingCode = '';
    if (data.isNewCustomer && !(editInfo && editInfo.startRow)) {
      const pr = createPendingCustomerInternal_(data);
      if (pr.error) return { success: false, message: pr.error };
      if (pr.pending) { usedPending = true; pendingCode = pr.code; }
      else { data.customerCode = pr.code; }         // khách đã có chính thức (trùng SĐT) -> dùng mã sẵn có
    } else if (data.customerCode && _pendFindByCode_(data.customerCode)) {
      usedPending = true; pendingCode = _normCode_(data.customerCode);   // khách chờ duyệt: chưa ghi vào DataKhachSi
    }

    _appendOrderRows(sh, data);

    const customerCode = usedPending ? pendingCode : _saveNewCustomerInternal(data.name, data.phone, data.address, data.priceType,
      data.city || '', data.taxCode || '', data.company || '', data.invoiceName || '',
      { customerCode: data.customerCode || '', codeGroup: data.codeGroup || '', codeFactory: data.codeFactory || '', cccd: data.cccd || '' });

    if (isDebt) {
      const ref = 'DH_' + new Date().getTime();
      _saveCongNoInternal(ref, data.name, data.phone, new Date().toISOString(), data.totalAmount, data.employeeName, data.priceType);
    }

    // [MỚI] cập nhật tổng hợp theo tháng - ghi nhẹ, không quét lại DonHang
    try {
      const newQty = (data.cartDetails || []).reduce((a, it) => a + (Number(it.qty) || 0), 0);
      let newDate = new Date();
      if (data.orderDate) { const d = new Date(data.orderDate); if (!isNaN(d.getTime())) newDate = d; }
      if (oldTotals && oldCounted) {
        _tongHopApplyDelta(oldTotals.date, -oldTotals.revenue, -1, -oldTotals.qty, oldTotals.phone);
      }
      // Đơn mới từ 01/10/2026 chưa được duyệt nên CHƯA cộng vào doanh thu; chỉ cộng khi bấm Duyệt.
      if (!_orderNeedsApproval(newDate)) {
        _tongHopApplyDelta(newDate, Number(data.totalAmount) || 0, 1, newQty, data.phone);
      }
    } catch (e2) { /* không chặn luồng chính nếu lỗi ghi tổng hợp */ }

    if (dupKey) { try { CacheService.getScriptCache().put(dupKey, '1', 180); } catch (e3) {} }
    let invoiceId = '';
    try { invoiceId = _invCreateForOrder_(data, customerCode || '', Number(data.totalAmount) || 0); } catch (e4) { Logger.log('Không tạo được dòng Theo dõi xuất HĐ: ' + e4); }
    return { success: true, customerCode: customerCode || '', invoiceId: invoiceId, pendingApproval: usedPending, message: editInfo && editInfo.startRow ? 'Đã hủy đơn cũ và lưu đơn thay thế thành công!' : (usedPending ? 'Đơn đã lưu. Khách mới đang chờ Admin/QLB2B duyệt.' : 'Đơn hàng đã lưu thành công!') };
  } catch (e) {
    throw new Error('Lỗi khi lưu đơn hàng: ' + e.toString());
  } finally {
    lock.releaseLock();
  }
}

function cancelOrderOnly(startRow, rowCount, employeeName, fp) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch (e) { throw new Error('Hệ thống đang bận, thử lại sau.'); }
  try {
    const sh = _donHangSheet();
    const chk = _wgCheckBlock_(sh, startRow, rowCount, fp);
    if (!chk.ok) return 'Lỗi: ' + chk.error;
    const totals = _readOrderBlockTotals(sh, startRow, rowCount);
    const wasCounted = _isBlockCounted(sh, startRow);
    const note = 'Đã HỦY HẲN lúc ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm dd/MM/yyyy') +
      (employeeName ? ' bởi ' + employeeName : '') + ' (không tạo đơn thay thế).';
    _cancelOrderRows(sh, startRow, rowCount, note);
    try {
      if (totals && wasCounted) _tongHopApplyDelta(totals.date, -totals.revenue, -1, -totals.qty, totals.phone);
    } catch (e2) {}
    return 'Đã hủy đơn hàng (không tạo đơn thay thế)!';
  } catch (e) {
    return 'Lỗi: ' + e.toString();
  } finally { lock.releaseLock(); }
}

//  THAY THẾ: getBusinessDashboard (đọc từ TongHopThang thay vì
//  quét toàn bộ 1 năm DonHang - CÙNG CẤU TRÚC TRẢ VỀ NHƯ CŨ, front-end
//  index.html KHÔNG cần sửa gì)
// ════════════════════════════════════════════════════════════
function getBusinessDashboard() {
  try {
    const now = new Date();
    const y = now.getFullYear(), m = now.getMonth();

    function startOfMonth(yr, mo) { return new Date(yr, mo, 1, 0, 0, 0, 0); }
    function endOfMonth(yr, mo) { return new Date(yr, mo + 1, 0, 23, 59, 59, 999); }
    function pct(cur, prev) { if (!prev) return cur ? 100 : 0; return (cur - prev) / prev * 100; }
    function mkey(yr, mo) { return yr + '-' + ('0' + (mo + 1)).slice(-2); }
    function emptyStat() { return { revenue: 0, orders: 0, qty: 0, aov: 0, customers: 0 }; }

    const tongHop = _loadTongHopMap();
    function statFromMap(key) { return tongHop[key] ? Object.assign({}, tongHop[key]) : emptyStat(); }
    function sumStats(keys) {
      let revenue = 0, orders = 0, qty = 0;
      keys.forEach(k => { const s = tongHop[k]; if (s) { revenue += s.revenue; orders += s.orders; qty += s.qty; } });
      return { revenue: revenue, orders: orders, qty: qty, aov: orders ? Math.round(revenue / orders) : 0, customers: 0 };
    }

    const thisMonthKey = mkey(y, m);
    let lmY = y, lmM = m - 1; if (lmM < 0) { lmM = 11; lmY--; }
    const lastMonthKey = mkey(lmY, lmM);

    const q = Math.floor(m / 3);
    const qMonths = [q * 3, q * 3 + 1, q * 3 + 2].map(mo => mkey(y, mo));
    let lq = q - 1, lqY = y; if (lq < 0) { lq = 3; lqY--; }
    const lqMonths = [lq * 3, lq * 3 + 1, lq * 3 + 2].map(mo => mkey(lqY, mo));

    const ytdMonths = []; for (let mo = 0; mo <= m; mo++) ytdMonths.push(mkey(y, mo));
    const ytdLastMonths = []; for (let mo = 0; mo <= m; mo++) ytdLastMonths.push(mkey(y - 1, mo));

    let thisMonth = statFromMap(thisMonthKey);
    let lastMonth = statFromMap(lastMonthKey);
    const thisQuarter = sumStats(qMonths);
    const lastQuarter = sumStats(lqMonths);
    const ytdThis = sumStats(ytdMonths);
    const ytdLast = sumStats(ytdLastMonths);

    const monthlySeries = [];
    for (let i = 11; i >= 0; i--) {
      let yy = y, mm = m - i; while (mm < 0) { mm += 12; yy--; }
      const key = mkey(yy, mm);
      const s = statFromMap(key);
      monthlySeries.push({ label: ('0' + (mm + 1)).slice(-2) + '/' + yy, revenue: s.revenue, orders: s.orders });
    }
    const seriesWithData = monthlySeries.filter(x => x.revenue > 0);
    const seriesAvg = seriesWithData.length ? seriesWithData.reduce((a, x) => a + x.revenue, 0) / seriesWithData.length : 0;
    monthlySeries.forEach(x => { x.pctVsAvg = seriesAvg ? ((x.revenue - seriesAvg) / seriesAvg * 100) : 0; });

    const cumThis = [], cumLast = [];
    let runThis = 0, runLast = 0;
    for (let mo = 0; mo < 12; mo++) {
      const keyThis = mkey(y, mo), keyLast = mkey(y - 1, mo);
      const sT = startOfMonth(y, mo);
      if (sT <= now) { runThis += statFromMap(keyThis).revenue; cumThis.push(runThis); } else { cumThis.push(null); }
      runLast += statFromMap(keyLast).revenue; cumLast.push(runLast);
    }

    // Chi tiết THÁNG NÀY - chỉ đọc DonHang trong phạm vi tháng hiện tại (nhẹ, không phình theo thời gian)
    const thisMonthStart = startOfMonth(y, m), thisMonthEnd = endOfMonth(y, m);
    const thisMonthOrders = _readDonHangGrouped(thisMonthStart.toISOString(), thisMonthEnd.toISOString(), null, true);
    const byEmp = {}, byPriceType = {}, byProduct = {};
    thisMonthOrders.forEach(o => {
      if (!byEmp[o.emp]) byEmp[o.emp] = { orders: 0, revenue: 0 };
      byEmp[o.emp].orders++; byEmp[o.emp].revenue += o.total;
      if (!byPriceType[o.priceType]) byPriceType[o.priceType] = { orders: 0, revenue: 0 };
      byPriceType[o.priceType].orders++; byPriceType[o.priceType].revenue += o.total;
      o.items.forEach(it => {
        if (!byProduct[it.name]) byProduct[it.name] = { qty: 0, revenue: 0 };
        byProduct[it.name].qty += it.qty; byProduct[it.name].revenue += it.price;
      });
    });
    const empArr = Object.keys(byEmp).map(k => ({ name: k, orders: byEmp[k].orders, revenue: byEmp[k].revenue })).sort((a, b) => b.revenue - a.revenue);
    const priceTypeArr = Object.keys(byPriceType).map(k => ({ name: k, orders: byPriceType[k].orders, revenue: byPriceType[k].revenue })).sort((a, b) => b.revenue - a.revenue);
    const productArr = Object.keys(byProduct).map(k => ({ name: k, qty: byProduct[k].qty, revenue: byProduct[k].revenue })).sort((a, b) => b.revenue - a.revenue).slice(0, 10);

    // Doanh thu nhân viên THÁNG TRƯỚC - chỉ đọc đúng 1 tháng (phục vụ cảnh báo so sánh)
    let lmY2 = y, lmM2 = m - 1; if (lmM2 < 0) { lmM2 = 11; lmY2--; }
    const lastMonthStart = startOfMonth(lmY2, lmM2), lastMonthEnd = endOfMonth(lmY2, lmM2);
    const lastMonthOrders = _readDonHangGrouped(lastMonthStart.toISOString(), lastMonthEnd.toISOString(), null, true);

    // [FIX v6.2] Tự phục hồi nếu TongHopThang trống/chưa đồng bộ - tính lại
    // trực tiếp từ DonHang cho tháng này & tháng trước, đảm bảo KPI luôn
    // hiển thị đúng số liệu ngay cả khi chưa chạy rebuildTongHopThang().
    if (thisMonth.revenue === 0 && thisMonthOrders.length > 0) {
      let rev=0, qty=0; const cust={};
      thisMonthOrders.forEach(o=>{ rev+=o.total; o.items.forEach(it=>qty+=it.qty); if(o.phone) cust[o.phone]=1; });
      thisMonth = { revenue: rev, orders: thisMonthOrders.length, qty: qty, aov: Math.round(rev/thisMonthOrders.length), customers: Object.keys(cust).length };
    }
    if (lastMonth.revenue === 0 && lastMonthOrders.length > 0) {
      let rev=0, qty=0; const cust={};
      lastMonthOrders.forEach(o=>{ rev+=o.total; o.items.forEach(it=>qty+=it.qty); if(o.phone) cust[o.phone]=1; });
      lastMonth = { revenue: rev, orders: lastMonthOrders.length, qty: qty, aov: Math.round(rev/lastMonthOrders.length), customers: Object.keys(cust).length };
    }

    const byEmpLast = {};
    lastMonthOrders.forEach(o => { byEmpLast[o.emp] = (byEmpLast[o.emp] || 0) + o.total; });

    const lastOrderDate = _getLastOrderDate();
    const daysSinceLastOrder = lastOrderDate ? Math.floor((now - lastOrderDate) / (1000 * 60 * 60 * 24)) : null;

    const warnings = [];
    const momPct = pct(thisMonth.revenue, lastMonth.revenue);
    if (lastMonth.revenue > 0) {
      if (momPct <= -15) warnings.push({ level: 'danger', text: 'Doanh thu tháng này giảm mạnh ' + Math.abs(momPct).toFixed(1) + '% so với tháng trước - cần rà soát ngay nguồn khách, chính sách giá và hoạt động của đối thủ.' });
      else if (momPct <= -5) warnings.push({ level: 'warning', text: 'Doanh thu tháng này giảm ' + Math.abs(momPct).toFixed(1) + '% so với tháng trước - cần theo dõi sát trong các ngày tới.' });
      else if (momPct >= 20) warnings.push({ level: 'success', text: 'Doanh thu tháng này tăng trưởng tốt, +' + momPct.toFixed(1) + '% so với tháng trước.' });
    }
    const qoqPct = pct(thisQuarter.revenue, lastQuarter.revenue);
    if (lastQuarter.revenue > 0 && qoqPct <= -10) warnings.push({ level: 'warning', text: 'Doanh thu quý này giảm ' + Math.abs(qoqPct).toFixed(1) + '% so với quý trước.' });

    const aovPct = pct(thisMonth.aov, lastMonth.aov);
    if (lastMonth.aov > 0 && aovPct <= -10) warnings.push({ level: 'warning', text: 'Giá trị đơn hàng trung bình (AOV) giảm ' + Math.abs(aovPct).toFixed(1) + '% so với tháng trước - khách đang mua ít hơn mỗi lần đặt.' });

    const orderCountPct = pct(thisMonth.orders, lastMonth.orders);
    if (lastMonth.orders > 0 && orderCountPct <= -15) warnings.push({ level: 'warning', text: 'Số lượng đơn hàng giảm ' + Math.abs(orderCountPct).toFixed(1) + '% so với tháng trước - tốc độ đặt hàng đang chậm lại.' });

    if (daysSinceLastOrder !== null && daysSinceLastOrder >= 3) warnings.push({ level: 'danger', text: 'Đã ' + daysSinceLastOrder + ' ngày hệ thống chưa ghi nhận đơn hàng mới - kiểm tra ngay kênh bán hàng / trang đặt hàng.' });

    Object.keys(byEmpLast).forEach(name => {
      const cur = (byEmp[name] ? byEmp[name].revenue : 0);
      const prev = byEmpLast[name];
      if (prev >= 500000) {
        const p = pct(cur, prev);
        if (p <= -30) warnings.push({ level: 'warning', text: 'Nhân viên ' + name + ' có doanh thu tháng này giảm ' + Math.abs(p).toFixed(1) + '% so với tháng trước.' });
      }
    });

    const yoyPct = pct(ytdThis.revenue, ytdLast.revenue);
    if (ytdLast.revenue > 0 && yoyPct <= -10) warnings.push({ level: 'danger', text: 'Lũy kế năm nay đang thấp hơn ' + Math.abs(yoyPct).toFixed(1) + '% so với cùng kỳ năm trước.' });
    else if (ytdLast.revenue > 0 && yoyPct >= 15) warnings.push({ level: 'success', text: 'Lũy kế năm nay đang cao hơn cùng kỳ năm trước ' + yoyPct.toFixed(1) + '% - đà tăng trưởng tốt.' });

    if (!warnings.length) warnings.push({ level: 'success', text: 'Không phát hiện dấu hiệu bất thường - tình hình kinh doanh đang ổn định.' });

    return {
      thisMonth: thisMonth, lastMonth: lastMonth, momPct: momPct,
      thisQuarter: thisQuarter, lastQuarter: lastQuarter, qoqPct: qoqPct,
      ytdThis: ytdThis, ytdLast: ytdLast, yoyPct: yoyPct,
      monthlySeries: monthlySeries, seriesAvg: seriesAvg, cumThis: cumThis, cumLast: cumLast,
      empArr: empArr, priceTypeArr: priceTypeArr, productArr: productArr,
      daysSinceLastOrder: daysSinceLastOrder, warnings: warnings,
      generatedAt: now.toISOString()
    };
  } catch (e) {
    return { error: e.toString() };
  }
}

}

function _loadCustomerFirstOrderMap() {
  const sh = _dataKhachSiSheet();
  const map = {};
  if (!sh || sh.getLastRow() < 2) return map;
  const data = sh.getRange(2, 1, sh.getLastRow() - 1, 18).getValues();
  data.forEach(r => {
    const phone = (r[DKS_COL.PHONE - 1] || '').toString().replace(/\D/g, '');
    const first = r[DKS_COL.FIRST_ORDER - 1];
    if (phone && first) {
      const d = first instanceof Date ? first : new Date(first);
      if (!isNaN(d.getTime())) map[phone] = d;
    }
  });
  return map;
}

function _loadCustomerManualSegMap() {
  const sh = _dataKhachSiSheet();
  const map = {};
  if (!sh || sh.getLastRow() < 2) return map;
  const lastRow = sh.getLastRow();
  const data = sh.getRange(2, 1, lastRow - 1, DKS_COL_MANUAL_SEG).getValues();
  data.forEach(r => {
    const phone = (r[DKS_COL.PHONE - 1] || '').toString().replace(/\D/g, '');
    const seg = (r[DKS_COL_MANUAL_SEG - 1] || '').toString().trim();
    if (phone && seg) map[phone] = seg;
  });
  return map;
}

// [MỚI v6.2] Đánh dấu thủ công 1 khách hàng là Khách Cũ / Khách Mới
function setCustomerManualSegment(phone, name, segType) {
  try {
    const sh = _dataKhachSiSheet();
    if (!sh) return { success: false, error: 'Không tìm thấy sheet DataKhachSi.' };
    const pc = (phone || '').toString().replace(/[^0-9]/g, '');
    if (!pc) return { success: false, error: 'Thiếu số điện thoại khách hàng.' };
    const seg = (segType === 'Cu' || segType === 'Moi') ? segType : '';
    const lastRow = sh.getLastRow();
    const data = lastRow > 1 ? sh.getRange(2, 1, lastRow - 1, 2).getValues() : [];
    for (let i = 0; i < data.length; i++) {
      const rowPhone = (data[i][DKS_COL.PHONE - 1] || '').toString().replace(/[^0-9]/g, '');
      if (rowPhone === pc) {
        sh.getRange(i + 2, DKS_COL_MANUAL_SEG).setValue(seg);
        return { success: true };
      }
    }
    const newRow = sh.getLastRow() + 1;
    const arr = new Array(DKS_COL_MANUAL_SEG).fill('');
    arr[DKS_COL.NAME - 1] = name || '';
    arr[DKS_COL.PHONE - 1] = pc;
    arr[DKS_COL_MANUAL_SEG - 1] = seg;
    sh.getRange(newRow, 2).setNumberFormat('@');
    sh.getRange(newRow, 1, 1, DKS_COL_MANUAL_SEG).setValues([arr]);
    return { success: true };
  } catch (e) { return { success: false, error: e.toString() }; }
}

function getCustomerSegmentsByEmployee(fromDate, toDate) {
  try {
    const now = new Date();
    const from = fromDate ? new Date(fromDate) : new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
    const to = toDate ? new Date(toDate) : now;
    to.setHours(23, 59, 59, 999);
    const orders = _readDonHangGrouped(from.toISOString(), to.toISOString(), null, true);
    const firstOrderMap = _loadCustomerFirstOrderMap();
    const manualMap = _loadCustomerManualSegMap(); // [MỚI] ghi đè phân loại tự động

    const empMap = {};
    orders.forEach(o => {
      const phone = (o.phone || '').toString().replace(/\D/g, '');
      if (!phone) return;
      const emp = o.emp || '(Không rõ)';
      const manual = manualMap[phone] || '';
      let seg;
      if (manual === 'Cu') seg = 'old';
      else if (manual === 'Moi') seg = 'new';
      else {
        const firstOrder = firstOrderMap[phone];
        const isOld = firstOrder ? (firstOrder < from) : false;
        seg = isOld ? 'old' : 'new';
      }
      if (!empMap[emp]) empMap[emp] = { old: {}, new: {} };
      if (!empMap[emp][seg][phone]) empMap[emp][seg][phone] = { name: o.name, phone: o.phone, revenue: 0, orders: 0, manualSeg: manual };
      empMap[emp][seg][phone].revenue += o.total;
      empMap[emp][seg][phone].orders += 1;
    });

    return Object.keys(empMap).map(emp => {
      const oldList = Object.values(empMap[emp].old).sort((a, b) => b.revenue - a.revenue);
      const newList = Object.values(empMap[emp].new).sort((a, b) => b.revenue - a.revenue);
      return {
        employee: emp,
        oldCount: oldList.length, newCount: newList.length,
        oldCustomers: oldList, newCustomers: newList
      };
    }).sort((a, b) => (b.oldCount + b.newCount) - (a.oldCount + a.newCount));
  } catch (e) { return []; }
}

function getCustomerRankingByMonth(monthKey) {
  try {
    const parts = (monthKey || '').split('-');
    if (parts.length !== 2) return { month: monthKey, byRevenue: [], byOrders: [] };
    const y = parseInt(parts[0]), m = parseInt(parts[1]) - 1;
    const from = new Date(y, m, 1, 0, 0, 0, 0);
    const to = new Date(y, m + 1, 0, 23, 59, 59, 999);
    const orders = _readDonHangGrouped(from.toISOString(), to.toISOString(), null, true);

    const byCust = {};
    orders.forEach(o => {
      const key = (o.phone || o.name || 'Khách lẻ').toString();
      if (!byCust[key]) byCust[key] = { name: o.name, phone: o.phone, orders: 0, revenue: 0 };
      byCust[key].orders++;
      byCust[key].revenue += o.total;
    });

    const byRevenue = Object.values(byCust).sort((a, b) => b.revenue - a.revenue).slice(0, 20);
    const byOrders = Object.values(byCust).slice().sort((a, b) => b.orders - a.orders).slice(0, 20);
    return { month: monthKey, byRevenue: byRevenue, byOrders: byOrders };
  } catch (e) { return { month: monthKey, byRevenue: [], byOrders: [], error: e.toString() }; }
}

// ════════════════════════════════════════════════════════════
//  THAY THẾ getBusinessDashboard(): thêm avgPerDay + projected
//  cho mỗi tháng trong monthlySeries (TB doanh thu/ngày + dự kiến).
//  Phần còn lại giữ nguyên logic đọc từ TongHopThang như bản trước.
// ════════════════════════════════════════════════════════════
function getBusinessDashboard() {
  try {
    const now = new Date();
    const y = now.getFullYear(), m = now.getMonth();

    function startOfMonth(yr, mo) { return new Date(yr, mo, 1, 0, 0, 0, 0); }
    function endOfMonth(yr, mo) { return new Date(yr, mo + 1, 0, 23, 59, 59, 999); }
    function pct(cur, prev) { if (!prev) return cur ? 100 : 0; return (cur - prev) / prev * 100; }
    function mkey(yr, mo) { return yr + '-' + ('0' + (mo + 1)).slice(-2); }
    function emptyStat() { return { revenue: 0, orders: 0, qty: 0, aov: 0, customers: 0 }; }
    function daysInMonth(yr, mo) { return new Date(yr, mo + 1, 0).getDate(); }

    const tongHop = _loadTongHopMap();
    function statFromMap(key) { return tongHop[key] ? Object.assign({}, tongHop[key]) : emptyStat(); }
    function sumStats(keys) {
      let revenue = 0, orders = 0, qty = 0;
      keys.forEach(k => { const s = tongHop[k]; if (s) { revenue += s.revenue; orders += s.orders; qty += s.qty; } });
      return { revenue: revenue, orders: orders, qty: qty, aov: orders ? Math.round(revenue / orders) : 0, customers: 0 };
    }

    const thisMonthKey = mkey(y, m);
    let lmY = y, lmM = m - 1; if (lmM < 0) { lmM = 11; lmY--; }
    const lastMonthKey = mkey(lmY, lmM);

    const q = Math.floor(m / 3);
    const qMonths = [q * 3, q * 3 + 1, q * 3 + 2].map(mo => mkey(y, mo));
    let lq = q - 1, lqY = y; if (lq < 0) { lq = 3; lqY--; }
    const lqMonths = [lq * 3, lq * 3 + 1, lq * 3 + 2].map(mo => mkey(lqY, mo));

    const ytdMonths = []; for (let mo = 0; mo <= m; mo++) ytdMonths.push(mkey(y, mo));
    const ytdLastMonths = []; for (let mo = 0; mo <= m; mo++) ytdLastMonths.push(mkey(y - 1, mo));

    const thisMonth = statFromMap(thisMonthKey);
    const lastMonth = statFromMap(lastMonthKey);
    const thisQuarter = sumStats(qMonths);
    const lastQuarter = sumStats(lqMonths);
    const ytdThis = sumStats(ytdMonths);
    const ytdLast = sumStats(ytdLastMonths);

    const monthlySeries = [];
    for (let i = 11; i >= 0; i--) {
      let yy = y, mm = m - i; while (mm < 0) { mm += 12; yy--; }
      const key = mkey(yy, mm);
      const s = statFromMap(key);
      const dim = daysInMonth(yy, mm);
      const isCurrentMonth = (yy === y && mm === m);
      const daysElapsed = isCurrentMonth ? now.getDate() : dim;
      const avgPerDay = daysElapsed ? (s.revenue / daysElapsed) : 0;
      const projected = isCurrentMonth ? avgPerDay * dim : s.revenue;
      monthlySeries.push({
        label: ('0' + (mm + 1)).slice(-2) + '/' + yy, revenue: s.revenue, orders: s.orders,
        avgPerDay: avgPerDay, projected: projected
      });
    }
    const seriesWithData = monthlySeries.filter(x => x.revenue > 0);
    const seriesAvg = seriesWithData.length ? seriesWithData.reduce((a, x) => a + x.revenue, 0) / seriesWithData.length : 0;
    monthlySeries.forEach(x => { x.pctVsAvg = seriesAvg ? ((x.revenue - seriesAvg) / seriesAvg * 100) : 0; });

    const cumThis = [], cumLast = [];
    let runThis = 0, runLast = 0;
    for (let mo = 0; mo < 12; mo++) {
      const keyThis = mkey(y, mo), keyLast = mkey(y - 1, mo);
      const sT = startOfMonth(y, mo);
      if (sT <= now) { runThis += statFromMap(keyThis).revenue; cumThis.push(runThis); } else { cumThis.push(null); }
      runLast += statFromMap(keyLast).revenue; cumLast.push(runLast);
    }

    const thisMonthStart = startOfMonth(y, m), thisMonthEnd = endOfMonth(y, m);
    const thisMonthOrders = _readDonHangGrouped(thisMonthStart.toISOString(), thisMonthEnd.toISOString(), null, true);
    const byEmp = {}, byPriceType = {}, byProduct = {};
    thisMonthOrders.forEach(o => {
      if (!byEmp[o.emp]) byEmp[o.emp] = { orders: 0, revenue: 0 };
      byEmp[o.emp].orders++; byEmp[o.emp].revenue += o.total;
      if (!byPriceType[o.priceType]) byPriceType[o.priceType] = { orders: 0, revenue: 0 };
      byPriceType[o.priceType].orders++; byPriceType[o.priceType].revenue += o.total;
      o.items.forEach(it => {
        if (!byProduct[it.name]) byProduct[it.name] = { qty: 0, revenue: 0 };
        byProduct[it.name].qty += it.qty; byProduct[it.name].revenue += it.price;
      });
    });
    const empArr = Object.keys(byEmp).map(k => ({ name: k, orders: byEmp[k].orders, revenue: byEmp[k].revenue })).sort((a, b) => b.revenue - a.revenue);
    const priceTypeArr = Object.keys(byPriceType).map(k => ({ name: k, orders: byPriceType[k].orders, revenue: byPriceType[k].revenue })).sort((a, b) => b.revenue - a.revenue);
    const productArr = Object.keys(byProduct).map(k => ({ name: k, qty: byProduct[k].qty, revenue: byProduct[k].revenue })).sort((a, b) => b.revenue - a.revenue).slice(0, 10);

    let lmY2 = y, lmM2 = m - 1; if (lmM2 < 0) { lmM2 = 11; lmY2--; }
    const lastMonthStart = startOfMonth(lmY2, lmM2), lastMonthEnd = endOfMonth(lmY2, lmM2);
    const lastMonthOrders = _readDonHangGrouped(lastMonthStart.toISOString(), lastMonthEnd.toISOString(), null, true);
    const byEmpLast = {};
    lastMonthOrders.forEach(o => { byEmpLast[o.emp] = (byEmpLast[o.emp] || 0) + o.total; });

    const lastOrderDate = _getLastOrderDate();
    const daysSinceLastOrder = lastOrderDate ? Math.floor((now - lastOrderDate) / (1000 * 60 * 60 * 24)) : null;

    const warnings = [];
    const momPct = pct(thisMonth.revenue, lastMonth.revenue);
    if (lastMonth.revenue > 0) {
      if (momPct <= -15) warnings.push({ level: 'danger', text: 'Doanh thu tháng này giảm mạnh ' + Math.abs(momPct).toFixed(1) + '% so với tháng trước - cần rà soát ngay nguồn khách, chính sách giá và hoạt động của đối thủ.' });
      else if (momPct <= -5) warnings.push({ level: 'warning', text: 'Doanh thu tháng này giảm ' + Math.abs(momPct).toFixed(1) + '% so với tháng trước - cần theo dõi sát trong các ngày tới.' });
      else if (momPct >= 20) warnings.push({ level: 'success', text: 'Doanh thu tháng này tăng trưởng tốt, +' + momPct.toFixed(1) + '% so với tháng trước.' });
    }
    const qoqPct = pct(thisQuarter.revenue, lastQuarter.revenue);
    if (lastQuarter.revenue > 0 && qoqPct <= -10) warnings.push({ level: 'warning', text: 'Doanh thu quý này giảm ' + Math.abs(qoqPct).toFixed(1) + '% so với quý trước.' });

    const aovPct = pct(thisMonth.aov, lastMonth.aov);
    if (lastMonth.aov > 0 && aovPct <= -10) warnings.push({ level: 'warning', text: 'Giá trị đơn hàng trung bình (AOV) giảm ' + Math.abs(aovPct).toFixed(1) + '% so với tháng trước - khách đang mua ít hơn mỗi lần đặt.' });

    const orderCountPct = pct(thisMonth.orders, lastMonth.orders);
    if (lastMonth.orders > 0 && orderCountPct <= -15) warnings.push({ level: 'warning', text: 'Số lượng đơn hàng giảm ' + Math.abs(orderCountPct).toFixed(1) + '% so với tháng trước - tốc độ đặt hàng đang chậm lại.' });

    if (daysSinceLastOrder !== null && daysSinceLastOrder >= 3) warnings.push({ level: 'danger', text: 'Đã ' + daysSinceLastOrder + ' ngày hệ thống chưa ghi nhận đơn hàng mới - kiểm tra ngay kênh bán hàng / trang đặt hàng.' });

    Object.keys(byEmpLast).forEach(name => {
      const cur = (byEmp[name] ? byEmp[name].revenue : 0);
      const prev = byEmpLast[name];
      if (prev >= 500000) {
        const p = pct(cur, prev);
        if (p <= -30) warnings.push({ level: 'warning', text: 'Nhân viên ' + name + ' có doanh thu tháng này giảm ' + Math.abs(p).toFixed(1) + '% so với tháng trước.' });
      }
    });

    const yoyPct = pct(ytdThis.revenue, ytdLast.revenue);
    if (ytdLast.revenue > 0 && yoyPct <= -10) warnings.push({ level: 'danger', text: 'Lũy kế năm nay đang thấp hơn ' + Math.abs(yoyPct).toFixed(1) + '% so với cùng kỳ năm trước.' });
    else if (ytdLast.revenue > 0 && yoyPct >= 15) warnings.push({ level: 'success', text: 'Lũy kế năm nay đang cao hơn cùng kỳ năm trước ' + yoyPct.toFixed(1) + '% - đà tăng trưởng tốt.' });

    if (!warnings.length) warnings.push({ level: 'success', text: 'Không phát hiện dấu hiệu bất thường - tình hình kinh doanh đang ổn định.' });

    return {
      thisMonth: thisMonth, lastMonth: lastMonth, momPct: momPct,
      thisQuarter: thisQuarter, lastQuarter: lastQuarter, qoqPct: qoqPct,
      ytdThis: ytdThis, ytdLast: ytdLast, yoyPct: yoyPct,
      monthlySeries: monthlySeries, seriesAvg: seriesAvg, cumThis: cumThis, cumLast: cumLast,
      empArr: empArr, priceTypeArr: priceTypeArr, productArr: productArr,
      daysSinceLastOrder: daysSinceLastOrder, warnings: warnings,
      generatedAt: now.toISOString()
    };
  } catch (e) {
    return { error: e.toString() };
  }
}

function _exportSSAsBase64Xlsx(tempSS) {
  SpreadsheetApp.flush();
  const url = 'https://docs.google.com/spreadsheets/d/' + tempSS.getId() + '/export?format=xlsx';
  const resp = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, muteHttpExceptions: true });

function exportTemCuonChieuExcel(labelsArr) {
  let tempSS = null;
  try {
    tempSS = SpreadsheetApp.create('TEMP_TEM_CUONCHIEU_' + new Date().getTime());
    const sh = tempSS.getSheets()[0];
    sh.setName('Tem Dán Thùng Dọc');
    sh.setHiddenGridlines(true); // ẩn lưới ô ly - nhìn như bản in thật, không cần chỉnh gì thêm
    sh.setColumnWidth(1, 175); // đủ rộng cho nhãn dài nhất "Nhiệt độ bảo quản"
    sh.setColumnWidth(2, 16);
    sh.setColumnWidth(3, 260); // thu hẹp lại - khung ôm sát chữ, không dư khoảng trắng
    sh.setColumnWidth(4, 30);
    const logoBlob = Utilities.newBlob(Utilities.base64Decode(MT_LOTTE_LOGO_BASE64), 'image/png', 'lotte_mart_logo.png');

    let row = 2;
    (labelsArr || []).forEach(function (item) {
      const n = Math.max(1, parseInt(item.boxCount) || 1);
      for (let i = 1; i <= n; i++) {
        _writeTemCuonChieuBlock(sh, row, item, i, n, logoBlob);
        row += 10; // đúng khoảng cách trong file mẫu (row 2,12,22,32...)
      }
    });
    return _exportSSAsBase64Xlsx(tempSS);
  } catch (e) {
    if (tempSS) { try { DriveApp.getFileById(tempSS.getId()).setTrashed(true); } catch (ex) {} }
    throw new Error('Lỗi xuất Excel tem cuốn chiếu: ' + e.toString());
  }
}
function _writeTemCuonChieuBlock(sh, startRow, item, boxIdx, boxTotal, logoBlob) {

    ['Thùng số: ' + boxIdx + '/' + boxTotal, '', ''],
    ['Nơi Giao Hàng', ':', item.storeName || ''],
    ['Mã NCC', ':', item.vendorCode || '010483'],
    ['Tên NCC', ':', item.ncc || 'CÔNG TY TNHH MỘT THÀNH VIÊN 9 SẠCH'],
    ['Ngành hàng', ':', item.nganhHang || 'DELICA'],
    ['Nhiệt độ bảo quản', ':', item.nhietDo || '-18 độ C'],
    ['Tổng số thùng', ':', String(boxTotal)]
  ];
  const nRows = rows.length;
  const nccRow = startRow + 3; // dòng "Tên NCC" - dòng thứ 4 trong khối (0-based index 3)

  sh.getRange(startRow + 1, 3, nRows - 1, 1).setNumberFormat('@');
  sh.getRange(startRow, 1, nRows, 3).setValues(rows);
  sh.getRange(startRow + 1, 3, nRows - 1, 1).setHorizontalAlignment('left');

function lookupTaxCode(taxCode) {
  try {
    var mst = (taxCode || '').toString().trim().replace(/[^0-9\-]/g, '');
    if (!mst) return { success: false, error: 'Vui lòng nhập mã số thuế.' };

    var searchUrl = 'https://masothue.com/Search/?q=' + encodeURIComponent(mst) + '&type=auto&force-search=1';
    var opts = {
      muteHttpExceptions: true,
      followRedirects: true,
      headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120 Safari/537.36' }
    };
    var resp = UrlFetchApp.fetch(searchUrl, opts);
    var html = resp.getContentText();

    // Nếu là trang danh sách kết quả (nhiều công ty trùng) -> lấy link đầu tiên rồi fetch trang chi tiết
    if (html.indexOf('itemprop="name"') === -1) {
      var linkMatch = html.match(/<h3[^>]*>\s*<a[^>]+href="(https:\/\/masothue\.com\/[^"]+)"/)
        || html.match(/<a[^>]+href="(https:\/\/masothue\.com\/[0-9][^"]*)"[^>]*>/);
      if (linkMatch) {
        var detailResp = UrlFetchApp.fetch(linkMatch[1], opts);
        html = detailResp.getContentText();
        searchUrl = linkMatch[1];
      }
    }

    function grab(re) {
      var m = html.match(re);
      return m ? m[1].replace(/<[^>]+>/g, ' ').replace(/&amp;/g, '&').replace(/&nbsp;/g, ' ').replace(/\s+/g, ' ').trim() : '';
    }

    var name = grab(/<h1[^>]*itemprop="name"[^>]*>([\s\S]*?)<\/h1>/i) || grab(/<h1[^>]*>([\s\S]*?)<\/h1>/i);
    var address = grab(/<[^>]+itemprop="address"[^>]*>([\s\S]*?)<\/(span|td|p)>/i)
      || grab(/Địa chỉ:?<\/(td|span|strong)>\s*<(td|span)[^>]*>([\s\S]*?)<\/\2>/i);
    var rep = grab(/Người đại diện:?<\/(td|span|strong)>\s*<(td|span)[^>]*>([\s\S]*?)<\/\2>/i);
    var status = grab(/Trạng thái:?<\/(td|span|strong)>\s*<(td|span)[^>]*>([\s\S]*?)<\/\2>/i);

    if (!name) {
      return { success: false, error: 'Không tìm thấy thông tin cho MST "' + mst + '" trên masothue.com. Vui lòng kiểm tra lại hoặc nhập tay.', link: searchUrl };
    }
    return { success: true, name: name, address: address, representative: rep, status: status, link: searchUrl };
  } catch (e) {
    return { success: false, error: 'Lỗi tra cứu: ' + e.toString(), link: 'https://masothue.com/' };
  }
}

function _nfc_(v) { return (v == null ? '' : v).toString().normalize('NFC'); }
function _normCode_(v) { return _nfc_(v).trim().toUpperCase(); }

function _provKeyList_() {
  const out = [];
  Object.keys(PROV_CODE_BY_NAME).forEach(k => out.push({ key: k, code: PROV_CODE_BY_NAME[k] }));
  Object.keys(PROV_ALIASES).forEach(k => out.push({ key: k, code: PROV_ALIASES[k] }));

// Tìm mã tỉnh trong 1 đoạn text (tên tỉnh hoặc cả địa chỉ); lấy tên tỉnh xuất hiện CUỐI CÙNG (tỉnh nằm cuối địa chỉ)
function _resolveProvCode_(text) {
  const s = ' ' + _stripDiacriticsServer(text || '').replace(/[^a-z0-9]+/g, ' ') + ' ';
  let best = '', bestIdx = -1;
  _provKeyList_().forEach(function (it) {
    const idx = s.lastIndexOf(' ' + it.key + ' ');
    if (idx > bestIdx) { bestIdx = idx; best = it.code; }
  });
  return best;
}
function _provCodeFromInput_(cityName, address) {
  return _resolveProvCode_(cityName) || _resolveProvCode_(address);
}
function _factoryForProv_(prov) {
  return FACTORY_CT_PROVS.indexOf(prov) !== -1 ? 'CT' : 'ĐN';
}
function _codePrefix_(factory, group, prov) {
  return (group === 'HCTM' || group === 'HCGC') ? factory + group : factory + group + prov;
}
function _nextSeq_(codes, prefix) {
  const p = _normCode_(prefix) + '-';
  let max = 0;
  (codes || []).forEach(function (c) {
    const v = _normCode_(c);
    if (v.indexOf(p) === 0) {
      const n = parseInt(v.substring(p.length), 10);
      if (!isNaN(n) && n > max) max = n;
    }
  });
  return max + 1;
}
// Sinh mã mới. Trả {code, factory, group, prov, error}
function _validProvCodes_() {
  const set = {};
  Object.keys(PROV_CODE_BY_NAME).forEach(k => { set[PROV_CODE_BY_NAME[k]] = true; });
  return Object.keys(set);
}
function _generateCode_(codes, group, factoryOverride, cityName, address, provOverride) {
  group = (group || '').toString().toUpperCase();
  if (CUST_CODE_GROUPS.indexOf(group) === -1) return { code: '', error: 'Nhóm khách không hợp lệ.' };
  let prov = '';
  const po = _nfc_(provOverride).trim().toUpperCase();
  if (po) {
    if (_validProvCodes_().indexOf(po) === -1) return { code: '', error: 'Mã tỉnh "' + po + '" không có trong danh mục 34 tỉnh/thành.' };
    prov = po;
  } else prov = _provCodeFromInput_(cityName, address);
  const isHC = (group === 'HCTM' || group === 'HCGC');
  if (!prov && !isHC) return { code: '', error: 'Chưa xác định được Tỉnh/Thành.' };
  let factory = _nfc_(factoryOverride).toUpperCase();
  if (factory !== 'CT' && factory !== 'ĐN') {
    // Hội chợ: toàn bộ mã Hội chợ đang dùng đều là xưởng ĐN -> mặc định ĐN; còn lại theo tỉnh
    factory = isHC ? 'ĐN' : _factoryForProv_(prov);
  }
  const prefix = _codePrefix_(factory, group, prov);
  let seq = _nextSeq_(codes, prefix);
  const digits = (group === 'CH') ? 3 : 4;
  const pad = n => ('0000' + n).slice(-digits);
  // Chốt chặn cuối: không bao giờ trả về mã đã tồn tại
  const taken = {};
  (codes || []).forEach(c => { const v = _normCode_(c); if (v) taken[v] = true; });
  while (taken[prefix + '-' + pad(seq)]) seq++;
  if (seq > Math.pow(10, digits) - 1) return { code: '', error: 'Đã hết số thứ tự cho nhóm ' + prefix + '.' };
  return { code: prefix + '-' + pad(seq), factory: factory, group: group, prov: prov, seq: seq };
}
function _readAllCodes_(sh) {

  let pend = [];
  try { pend = (typeof _pendCodes_ === 'function') ? _pendCodes_() : []; } catch (e) {}
  return base.concat(pend);
}

// [Client gọi] Cấp mã cho 1 khách CŨ chưa có mã (trang Data Khách Sỉ)
function assignCustomerCodeForRow(sheetRow, group, factory, manualCode, prov) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch (e) { return { success: false, error: 'Hệ thống đang bận, thử lại sau.' }; }
  try {
    const sh = _dataKhachSiSheet();
    const row = parseInt(sheetRow);
    if (!row || row < 2 || row > sh.getLastRow()) return { success: false, error: 'Dòng không hợp lệ.' };
    const cur = (sh.getRange(row, DKS_COL.CODE).getValue() || '').toString().trim();
    if (cur) return { success: false, error: 'Khách này đã có mã ' + cur + ' (mã đã cấp không được đổi).' };
    const address = (sh.getRange(row, DKS_COL.ADDRESS).getValue() || '').toString();
    let gen;
    if (group === 'ST' || group === 'XK') {
      const mc = _normCode_(manualCode);
      const mm = mc.match(/^(CT|ĐN)(ST|XK)[A-Z0-9ĐÀ-Ỹ]*-/);
      if (!mc || !mm || mm[2] !== group) return { success: false, error: 'Nhóm ' + group + ': nhập mã tay dạng CT' + group + 'XX-TÊN.' };
      if (_readAllCodes_(sh).some(c => _normCode_(c) === mc)) return { success: false, error: 'Mã ' + mc + ' đã tồn tại.' };
      gen = { code: mc, factory: mm[1], group: group, prov: '' };
    } else {
      gen = _generateCode_(_readAllCodes_(sh), group, factory, '', address, prov);
    }
    if (!gen.code) return { success: false, error: gen.error || 'Không tạo được mã.' };
    sh.getRange(row, DKS_COL.CODE).setNumberFormat('@');
    sh.getRange(row, DKS_COL.CODE, 1, 4).setValues([[gen.code, gen.factory, gen.prov, gen.group]]);
    return { success: true, code: gen.code, factory: gen.factory, prov: gen.prov, group: gen.group };
  } catch (e) {
    return { success: false, error: e.toString() };
  } finally { lock.releaseLock(); }
}

const NC_CODE_RE = /^(CT|ĐN)(HCTM|HCGC|DL\d|MN|CH|ST|XK|HC)/;

function _ncNormName_(s) { return _stripDiacriticsServer(s || '').replace(/[^a-z0-9]+/g, ' ').trim(); }
function _ncCleanName_(s) {
  s = (s == null ? '' : s).toString().trim();
  s = s.replace(/^bán\s+(hàng\s+)?cho\s+người\s+tiêu\s+dùng\s*[-–:]?\s*/i, '');
  s = s.replace(/\s*\((?:CCCD|CCDC|CMND)[^)]*\)\s*/ig, ' ').replace(/\s*\((?:điện thoại|dien thoai)[^)]*\)\s*/ig, ' ');
  return s.replace(/\s+/g, ' ').trim();
}
function _ncPhoneFromText_(s) {
  const m = (s || '').toString().match(/(?:điện thoại|dien thoai|sđt|sdt)[^\d]*(0\d{8,10})/i);
  return m ? m[1] : '';
}
// Cột "Mã khách hàng / MST / CCCD" trong file lẫn 3 loại: mã cũ KHSxxxx, CCCD (12 số, có thể mất số 0 đầu), MST (10 số hoặc 10-3 số)
function _ncClassifyId_(v) {
  const out = { tax: '', cccd: '', oldCode: '' };
  if (v == null || v === '') return out;
  let s = (typeof v === 'number') ? String(Math.round(v)) : v.toString();
  s = s.replace(/\s+/g, '');
  if (!s) return out;
  if (/^KHS\d+$/i.test(s)) { out.oldCode = s.toUpperCase(); return out; }
  if (/^[A-Za-zĐđ]/.test(s)) return out;                 // dạng CTXK-xxx: là mã, không phải số giấy tờ
  if (s.indexOf('-') !== -1) { out.tax = s; return out; }   // 0304741634-015
  let d = s.replace(/\D/g, '');
  if (d.length === 11) d = '0' + d;
  if (d.length === 12 || d.length === 9) out.cccd = d; else if (d) out.tax = d;
  return out;
}

function importNewCodeBatch(rows, opts) {
  opts = opts || {};
  const lock = LockService.getScriptLock();
  try { lock.waitLock(20000); } catch (e) { return { success: false, error: 'Hệ thống đang bận, thử lại sau.' }; }
  try {
    const sh = _dataKhachSiSheet();
    const W = DKS_TOTAL_COLS;
    const last = sh.getLastRow();
    const data = last > 1 ? sh.getRange(2, 1, last - 1, W).getValues() : [];
    const byCode = {}, byOld = {}, byPhone = {}, byIdName = {}, byNameAddr = {};
    const digits = v => (v == null ? '' : v).toString().replace(/[^0-9]/g, '');
    const index = function (r, i) {
      const c = _normCode_(r[DKS_COL.CODE - 1]); if (c) byCode[c] = i;
      const o = _normCode_(r[DKS_COL.OLD_CODE - 1]); if (o) byOld[o] = i;
      const p = digits(r[DKS_COL.PHONE - 1]); if (p) byPhone[p] = i;
      const nm = _ncNormName_(r[DKS_COL.NAME - 1]);
      const idn = digits(r[DKS_COL.TAX - 1]) || digits(r[DKS_COL.CCCD - 1]);
      if (idn && nm) byIdName[idn + '|' + nm] = i;
      if (nm) byNameAddr[nm + '|' + _ncNormName_(r[DKS_COL.ADDRESS - 1])] = i;
    };
    data.forEach(index);

    const changed = {};
    const added = [];
    const warnings = [];
    let nUpd = 0, nAdd = 0, nSkip = 0;

    (rows || []).forEach(function (x) {
      const rawName = (x.name || '').toString().trim();
      const alias = (x.alias || '').toString().trim();
      const newName = (x.newName || '').toString().trim();
      if (!rawName && !alias && !newName) { nSkip++; return; }

      let code = _normCode_(x.code);
      if (x.dupOf) { warnings.push('Mã ' + x.dupOf + ' bị trùng trong file: khách "' + (newName || alias || _ncCleanName_(rawName)) + '" được nhập KHÔNG mã (tab "chưa có mã") - hãy tự cấp mã.'); code = ''; }
      const id = _ncClassifyId_(x.idRaw);
      let phone = digits(x.phone) || digits(_ncPhoneFromText_(rawName));
      const m = code ? code.match(NC_CODE_RE) : null;
      let group = m ? m[2] : '';
      if (!group) { const g = (x.group || '').toString().trim().toUpperCase(); group = (g === 'HC') ? 'HC' : (/^DL\d?$/.test(g) ? (g === 'DL' ? 'DL1' : g) : (['MN', 'CH', 'ST', 'XK'].indexOf(g) !== -1 ? g : '')); }
      const factory = m ? m[1] : _nfc_(x.factory).trim().toUpperCase();
      let prov = _nfc_(x.prov).trim().toUpperCase();
      if (!prov && m && (group === 'MN' || group === 'CH' || /^DL\d$/.test(group))) prov = code.substr(m[0].length, 2);

      const cleanName = _ncCleanName_(rawName);
      const displayName = x.nameFromCol ? (cleanName || rawName) : (newName || alias || cleanName);   // nameFromCol (sheet DL): tên = đúng cột "Tên khách hàng"
      const nm = _ncNormName_(displayName), nmClean = _ncNormName_(cleanName);
      const addr = (x.address || '').toString().trim();
      const idn = id.tax.replace(/\D/g, '') || id.cccd;

      let idx = -1;
      if (code && byCode[code] != null) idx = byCode[code];
      if (idx < 0 && id.oldCode && byOld[id.oldCode] != null) idx = byOld[id.oldCode];
      if (idx < 0 && phone && byPhone[phone] != null) idx = byPhone[phone];
      if (idx < 0 && idn) idx = (byIdName[idn + '|' + nm] != null) ? byIdName[idn + '|' + nm] : (byIdName[idn + '|' + nmClean] != null ? byIdName[idn + '|' + nmClean] : -1);
      if (idx < 0) { const k = (byNameAddr[nm + '|' + _ncNormName_(addr)] != null) ? byNameAddr[nm + '|' + _ncNormName_(addr)] : byNameAddr[nmClean + '|' + _ncNormName_(addr)]; if (k != null) idx = k; }
      if (idx >= 0 && idx < data.length && code) {
        const cur = _normCode_(data[idx][DKS_COL.CODE - 1]);
        if (cur && cur !== code) idx = -1;   // khách đó đã mang mã khác → coi là người khác
      }

      const apply = function (arr) {
        const set = function (col, val) { if (val !== '' && val != null) arr[col - 1] = val; };
        set(DKS_COL.NAME, displayName);
        set(DKS_COL.ADDRESS, addr);
        set(DKS_COL.PHONE, phone);
        set(DKS_COL.INVOICE_NAME, rawName);
        set(DKS_COL.TAX, id.tax);
        set(DKS_COL.CCCD, id.cccd);
        set(DKS_COL.OLD_CODE, id.oldCode);
        if (alias && _ncNormName_(alias) !== _ncNormName_(displayName)) set(DKS_COL.DESCRIPTION, alias);
        if (code) { set(DKS_COL.CODE, code); set(DKS_COL.FACTORY, factory); set(DKS_COL.PROV, prov); set(DKS_COL.GROUP, group); }
        else if (group && !arr[DKS_COL.GROUP - 1]) set(DKS_COL.GROUP, group);
        return arr;
      };

      if (idx >= 0 && idx < data.length) {
        apply(data[idx]); changed[idx] = true; index(data[idx], idx); nUpd++;
      } else {
        const arr = apply(new Array(W).fill(''));
        data.push(arr); added.push(arr); index(arr, data.length - 1); nAdd++;
      }
    });

    const fmtCols = [DKS_COL.PHONE, DKS_COL.TAX, DKS_COL.CCCD, DKS_COL.CODE, DKS_COL.OLD_CODE];
    Object.keys(changed).forEach(function (k) {
      const i = parseInt(k, 10);
      if (i >= last - 1) return;           // dòng vừa thêm trong đợt này: ghi ở phần dưới
      fmtCols.forEach(c => sh.getRange(i + 2, c).setNumberFormat('@'));
      sh.getRange(i + 2, 1, 1, W).setValues([data[i]]);
    });
    if (added.length) {
      const start = sh.getLastRow() + 1;
      fmtCols.forEach(c => sh.getRange(start, c, added.length, 1).setNumberFormat('@'));
      sh.getRange(start, 1, added.length, W).setValues(added);
    }
    // dòng mới thêm nhưng bị "update" lại trong cùng đợt (cùng khách xuất hiện 2 lần) đã nằm trong `added` (cùng tham chiếu mảng) nên không mất dữ liệu
    return { success: true, updated: nUpd, added: nAdd, skipped: nSkip, warnings: warnings };
  } catch (e) {
    return { success: false, error: e.toString() };
  } finally { lock.releaseLock(); }
}

function getInvoiceTracking(opts) {
  try {
    opts = opts || {};
    const sh = _invSheet();
    const last = sh.getLastRow();
    if (last < 2) return [];
    const vals = sh.getRange(2, 1, last - 1, INV_HEADERS.length).getValues();
    const toks = _stripDiacriticsServer(opts.q || '').split(/\s+/).filter(Boolean);
    const out = [];
    for (let i = vals.length - 1; i >= 0; i--) {
      let o; try { o = _invRowObj_(vals[i], i + 2); } catch (eRow) { continue; }
      if (!o.id) continue;
      const cDay = o.created.slice(0, 10);
      if (opts.from && cDay < opts.from) continue;
      if (opts.to && cDay > opts.to) continue;
      if (opts.status && o.status !== opts.status) continue;
      if (opts.emp && o.emp !== opts.emp) continue;
      if (toks.length) {
        const hay = _stripDiacriticsServer([o.id, o.customer, o.code, o.address, o.emp, o.phone].join(' '));
        if (!toks.every(t => hay.indexOf(t) !== -1)) continue;
      }
      out.push(o);
      if (out.length >= (opts.limit || 600)) break;
    }
    return out;
  } catch (e) { return { error: e.toString() }; }
}

  return 0;
}

// Trigger 5 phút/lần: gửi tin đến hạn
function invTick_() {
  // B1: đọc KHÔNG khóa, chỉ lọc việc đến hạn
  const sh0 = _invSheet();
  const last0 = sh0.getLastRow();
  if (last0 < 2) return;
  const now = new Date();
  const due = [];
  sh0.getRange(2, 1, last0 - 1, INV_HEADERS.length).getValues().forEach(function (row, i) {
    const o = _invRowObj_(row, i + 2);
    if (!o.id) return;
    const open = (o.status === INV_ST.NONE || o.status === INV_ST.REQ || o.status === INV_ST.REQV);
    if (!open) return;
    const sendAt = _invParse_(o.sendAt), resendAt = _invParse_(o.resendAt);
    if (o.status === INV_ST.NONE && !o.sentAt && sendAt && now >= sendAt) due.push({ id: o.id, kind: 'send', o: o });
    else if (o.sentAt && resendAt && now >= resendAt) due.push({ id: o.id, kind: 'remind', o: o });
  });
  if (!due.length) return;   // không có gì đến hạn -> không đụng tới khóa

  // B2: gửi Zalo NGOÀI khóa
  due.forEach(function (d) {
    if (d.kind === 'send') {
      d.isVoid = (d.o.kind === 'Xuất bỏ');
      d.res = _invSendZalo_(_invMsgRequest_(d.o, d.isVoid));
    } else {
      d.res = _invSendZalo_(_invMsgRemind_(d.o));
    }
  });

  // B3: khóa NGẮN, chỉ ghi kết quả
  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (e) { return; }
  try {
    const sh = _invSheet();
    const t = new Date();
    due.forEach(function (d) {
      if (!d.res || !d.res.ok) return;
      const r = _invFindRow_(sh, d.id);
      if (!r) return;
      if (d.kind === 'send') {
        sh.getRange(r, IC.SENT).setNumberFormat('@').setValue(_invFmt_(t));
        sh.getRange(r, IC.SENDAT).setValue('');
        sh.getRange(r, IC.STATUS).setValue(d.isVoid ? INV_ST.REQV : INV_ST.REQ);
        sh.getRange(r, IC.STBY).setValue('Bot Zalo');
        sh.getRange(r, IC.STAT).setNumberFormat('@').setValue(_invFmt_(t));
      } else {
        sh.getRange(r, IC.RESEND).setValue('');
      }
    });
  } catch (e) { Logger.log('invTick_ lỗi: ' + e); } finally { lock.releaseLock(); }
}
function _invEnsureTrigger_() {
  const has = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'invTick_');
  if (!has) ScriptApp.newTrigger('invTick_').timeBased().everyMinutes(5).create();
}

}

function backfillInvoiceTrackingRun() {
  // Ví dụ lấy từ 01/10/2026 đến hôm nay:  return backfillInvoiceTracking('2026-10-01', _invToday_());
  return backfillInvoiceTracking(_invToday_(), _invToday_());

function backfillInvoiceTracking(fromStr, toStr) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const from = fromStr || _invToday_(), to = toStr || from;
    const dh = _donHangSheet();
    const last = dh.getLastRow();
    if (last < 2) { Logger.log('Chưa có đơn.'); return { added: 0 }; }
    const vals = dh.getRange(2, 1, last - 1, 23).getValues();

    // mã khách theo SĐT
    const codeByPhone = {};
    try {
      const dks = _dataKhachSiSheet();
      if (dks.getLastRow() > 1) dks.getRange(2, 1, dks.getLastRow() - 1, DKS_TOTAL_COLS).getValues().forEach(function (r) {
        const ph = (r[DKS_COL.PHONE - 1] || '').toString().replace(/[^0-9]/g, ''), cd = _normCode_(r[DKS_COL.CODE - 1]);
        if (ph && cd) codeByPhone[ph] = cd;
      });
    } catch (e) {}

    // dòng theo dõi đã có (để chống trùng)
    const ish = _invSheet(), il = ish.getLastRow();
    const exist = [];
    if (il > 1) ish.getRange(2, 1, il - 1, INV_HEADERS.length).getValues().forEach(function (r) {
      const o = _invRowObj_(r, 0), t = _invParse_(o.created);
      exist.push({ cust: o.customer.trim(), phone: o.phone.replace(/[^0-9]/g, ''), total: o.total, t: t ? t.getTime() : 0 });
    });

    // gom đơn: dòng có cột A (ngày) là dòng đầu của đơn, các dòng A trống bên dưới thuộc cùng đơn
    const blocks = [];
    let cur = null;
    vals.forEach(function (r) {
      if (r[0] !== '' && r[0] != null) { cur = { first: r, rows: [r] }; blocks.push(cur); }
      else if (cur && (r[4] || '').toString().trim()) cur.rows.push(r);
    });

    let added = 0, skippedDup = 0, skippedCancel = 0;
    blocks.forEach(function (b) {
      const f = b.first;
      const d = f[0] instanceof Date ? f[0] : new Date(f[0]);
      if (isNaN(d.getTime())) return;
      const day = Utilities.formatDate(d, _invTz_(), 'yyyy-MM-dd');
      if (day < from || day > to) return;
      if ((f[20] || '').toString() === 'Huy') { skippedCancel++; return; }
      let total = 0;
      const cart = [];
      b.rows.forEach(function (r) {
        const qty = Number(r[5]) || 0, val = Number(r[6]) || 0;
        if (!qty) return;
        total += val;
        const nm = (r[4] || '').toString().replace(/\s*\([^()]*\)\s*$/, '');
        cart.push({ name: nm, qty: qty, price: qty ? Math.round(val / qty) : 0 });
      });
      if (!cart.length) return;
      const phone = (f[2] || '').toString().replace(/[^0-9]/g, '');
      const name = (f[1] || '').toString();
      const dup = exist.some(function (x) {
        return x.cust === name.trim() && x.phone === phone && Math.round(x.total) === Math.round(total) && Math.abs(x.t - d.getTime()) < 60 * 60 * 1000;
      });
      if (dup) { skippedDup++; return; }
      const data = { name: name, phone: phone, address: (f[3] || '').toString(), taxCode: (f[9] || '').toString(), company: (f[10] || '').toString(),
        invoiceName: (f[22] || '').toString(), employeeName: (f[18] || '').toString(), orderDate: d, cartDetails: cart };
      _invCreateForOrder_(data, codeByPhone[phone] || '', total, { created: d });
      exist.push({ cust: name.trim(), phone: phone, total: total, t: d.getTime() });
      added++;
    });
    const msg = 'Bổ sung Theo dõi xuất HĐ (' + from + ' → ' + to + '): thêm ' + added + ' đơn, bỏ qua ' + skippedDup + ' đơn đã có, ' + skippedCancel + ' đơn đã hủy.';
    Logger.log(msg);
    return { added: added, skippedDup: skippedDup, skippedCancel: skippedCancel, message: msg };
  } finally { lock.releaseLock(); }
}

function getProvinceList() {

  return _validProvCodes_().map(c => ({ code: c, name: PROV_DISPLAY[c] || c, factory: _factoryForProv_(c), keys: keys[c] || [] }))

function _validCodeFormat_(code, group) {
  const c = _normCode_(code);
  const provs = _validProvCodes_().join('|');
  if (group === 'MN') return new RegExp('^(CT|ĐN)MN(' + provs + ')-\\d{4}$').test(c);
  if (group === 'CH') return new RegExp('^(CT|ĐN)CH(' + provs + ')-\\d{3,4}$').test(c);
  if (group === 'HCTM' || group === 'HCGC') return new RegExp('^(CT|ĐN)' + group + '-\\d{4}$').test(c);
  if (/^DL\d$/.test(group)) return new RegExp('^(CT|ĐN)' + group + '(' + provs + ')-\\d{3,4}$').test(c);
  if (group === 'ST' || group === 'XK') return new RegExp('^(CT|ĐN)' + group + '[A-Z0-9ĐÀ-Ỹ]+-[A-Z0-9ĐÀ-Ỹ]+$').test(c);
  return new RegExp('^(CT|ĐN)(HC)').test(c);
}

function auditCustomerCodes() {
  try {
    const sh = _dataKhachSiSheet();
    const last = sh.getLastRow();
    const data = last > 1 ? sh.getRange(2, 1, last - 1, DKS_TOTAL_COLS).getValues() : [];
    const rep = { total: data.length, withCode: 0, noCode: 0, duplicates: [], badFormat: [], mismatch: [], dupPhone: [], dupTax: [] };
    const byCode = {}, byPhone = {}, byTax = {};
    const dg = v => (v == null ? '' : v).toString().replace(/[^0-9]/g, '');
    data.forEach(function (r, i) {
      const row = i + 2, code = _normCode_(r[DKS_COL.CODE - 1]), name = (r[DKS_COL.NAME - 1] || '').toString();
      const ph = dg(r[DKS_COL.PHONE - 1]), tx = dg(r[DKS_COL.TAX - 1]);
      if (ph) (byPhone[ph] = byPhone[ph] || []).push(row);
      if (tx) (byTax[tx + '|' + _ncNormName_(name)] = byTax[tx + '|' + _ncNormName_(name)] || []).push(row);
      if (!code) { if (name || ph) rep.noCode++; return; }
      rep.withCode++;
      (byCode[code] = byCode[code] || []).push(row);
      const m = code.match(/^(CT|ĐN)(HCTM|HCGC|DL\d|MN|CH|ST|XK|HC)/);
      if (!m) { rep.badFormat.push({ row: row, code: code, name: name, why: 'Không bắt đầu bằng CT/ĐN + nhóm' }); return; }
      const grp = m[2];
      if (!_validCodeFormat_(code, grp)) rep.badFormat.push({ row: row, code: code, name: name, why: 'Sai cấu trúc ' + grp });
      const why = [];
      const gCol = _nfc_(r[DKS_COL.GROUP - 1]).trim().toUpperCase(), fCol = _nfc_(r[DKS_COL.FACTORY - 1]).trim().toUpperCase(), pCol = _nfc_(r[DKS_COL.PROV - 1]).trim().toUpperCase();
      if (gCol && gCol !== grp) why.push('cột Nhóm=' + gCol);
      if (fCol && fCol !== m[1]) why.push('cột Xưởng=' + fCol);
      if ((grp === 'MN' || grp === 'CH' || /^DL\d$/.test(grp))) {
        const pc = code.substr(m[0].length, 2);
        if (pCol && pCol !== pc) why.push('cột Tỉnh=' + pCol);
        if (_validProvCodes_().indexOf(pc) !== -1 && _factoryForProv_(pc) !== m[1]) why.push('xưởng ' + m[1] + ' không khớp tỉnh ' + pc + ' (đúng: ' + _factoryForProv_(pc) + ')');
      }
      if (why.length) rep.mismatch.push({ row: row, code: code, name: name, why: why.join('; ') });
    });
    Object.keys(byCode).forEach(k => { if (byCode[k].length > 1) rep.duplicates.push({ code: k, rows: byCode[k] }); });
    Object.keys(byPhone).forEach(k => { if (byPhone[k].length > 1) rep.dupPhone.push({ phone: k, rows: byPhone[k] }); });
    Object.keys(byTax).forEach(k => { if (byTax[k].length > 1) rep.dupTax.push({ key: k.split('|')[0], rows: byTax[k] }); });
    ['duplicates', 'badFormat', 'mismatch', 'dupPhone', 'dupTax'].forEach(k => { rep[k + 'Count'] = rep[k].length; rep[k] = rep[k].slice(0, 80); });
    rep.ok = !rep.duplicatesCount && !rep.badFormatCount && !rep.mismatchCount;
    return rep;
  } catch (e) { return { error: e.toString() }; }
}

function _hsv_(v) {
  if (v instanceof Date) return isNaN(v.getTime()) ? '' : Utilities.formatDate(v, 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy HH:mm');
  if (v === null || v === undefined) return '';
  return (typeof v === 'number' || typeof v === 'boolean') ? v : v.toString();
}
